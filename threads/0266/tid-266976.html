Comando que leva para o W.D.A.
pawn Code:
if(strcmp(cmdtext, "/wda", true) == 0)
{
    if(IsPlayerInAnyVehicle(playerid)) {
        new vehicleid = GetPlayerVehicleID(playerid);
        SetVehiclePos(vehicleid, -2565.3508, 1380.6610, 7.1875);
        SetVehicleZAngle(vehicleid, 359.9173);
        LinkVehicleToInterior(vehicleid, 0);
    }
    if(!IsPlayerInAnyVehicle(playerid)) {
        SetPlayerPos(playerid, -2509.4775, 1858.4069, 21.1858);
        SetPlayerFacingAngle(playerid, 269.6537);
        SetPlayerInterior(playerid, 0);
    }
    return 1;
}
