The above function does not check the file handle and actually there's no reason to open with a different mode just to "re-create" the file. io_append will do the job.

PHP Code:
AppendTo(filename[],string[])
{
    new File:myfile=fopen(filename,io_append);
    if (!myfile) return 0; // if file handle is invalid (0), the file couldn't be opened.
  
    new timestring[MAX_STRING];
    new hour,minu,seco,giorno,mese,anno;
    gettime(hour,minu,seco); getdate(anno,mese,giorno);
    format(timestring,sizeof(timestring),"[%d:%d:%d/%d-%d-%d]%s\n",hour,minu,seco,giorno,mese,anno,string);
    fwrite(myfile,timestring);
    fclose(myfile);
    return 1;
}
