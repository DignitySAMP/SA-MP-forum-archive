public OnPlayerStateChange(playerid, newstate, oldstate)
{

	{
		SendClientMessage(playerid, COLOR_GREY, "» Type /gps to use GPS system.");
	}
	return 1;
}

public OnPlayerEnterCheckpoint(playerid)
{
	SendClientMessage(playerid, COLOR_RED, "» You arrived destination..");
    DisablePlayerCheckpoint(playerid);
    return 1;
