I fixed,problem was in WHERE ID = `%d`

stock SavePlayer(playerid)
{
new query[200],Float: pos[4]; //Creates the variables
GetPlayerPos(playerid, pos[0], pos[1], pos[2]);
GetPlayerFacingAngle(playerid, pos[3]);
mysql_format(mysql, query, sizeof(query), "UPDATE `accounts` SET `Money` = %d, `Admin` = %d, `GM` = %d, `AdminKod` = %d, `GMKod` = %d, `VIP` = %d, `PosX` = %f, `PosY` = %f, `PosZ` = %f, `PosA` = %f WHERE `ID` = %d",
PlayerInfo[playerid][NovacDzep], PlayerInfo[playerid][Admin], PlayerInfo[playerid][GM], PlayerInfo[playerid][AdminKod], PlayerInfo[playerid][GMKod], PlayerInfo[playerid][Vip],pos[0], pos[1], pos[2], pos[3], PlayerInfo[playerid][ID]);

return 1;
