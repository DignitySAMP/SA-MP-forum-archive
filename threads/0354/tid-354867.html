Code:
dcmd_lbike(playerid,params[])
{
	#pragma unused params
 	if(AccInfo[playerid][Level] >= 2)
	{
		if (!IsPlayerInAnyVehicle(playerid))
		{
			CarSpawner(playerid,522);
			SendCommandToAdmins(playerid,"LBike");
			return SendClientMessage(playerid,yellow,"<<<<<<-Bike Successfully Created! ->>>>>");
		}
		else return SendClientMessage(playerid,red,"WARNING: You already have a vehicle");
	}
	else return WARNINGMessages(playerid, 7);
}

public OnPlayerSpawn(playerid)
{
HasSpawnedBike[playerid] = 0;
return 1;
}
new HasSpawnedBike[MAX_PLAYERS];
dcmd_lbike(playerid,params[])
{
    #pragma unused params
    if(AccInfo[playerid][Level] >= 2)
    {
        if (HasSpawnedBike[playerid] == 0)
        {
            CarSpawner(playerid,522);
            SendCommandToAdmins(playerid,"LBike");
                SendClientMessage(playerid,yellow,"<<<<<<-Bike Successfully Created! ->>>>>");
                        HasSpawnedBike[playerid] = 1;
        }
        else return SendClientMessage(playerid,red,"WARNING: You already have a vehicle");
    }
    else return WARNINGMessages(playerid, 7);
}
