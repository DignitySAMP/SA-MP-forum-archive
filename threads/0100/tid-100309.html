public OnPlayerStateChange(playerid, newstate, oldstate)
{
    new vehicle;
    new vehicleid;
    new string[128];

    if(newstate == 5)
    {
    vehicleid = GetPlayerVehicleID(playerid);
    vehicle = GetVehicleModel(vehicleid) - 400;
    format(string, sizeof(string), "~w~ %s", aVehicleNames[vehicle]);
    GameTextForPlayer(playerid, string, 3000, 1);
    }
    return 1;
}
To install it, simply put the codes above into your gamemode! Hope this is usefull!
