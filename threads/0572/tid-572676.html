BPS::CheckVehicle(playerid, vehicleid)
{
    if(GetPlayerVehicleID(playerid) == vehicleid &&
    GetPlayerState(playerid) == PLAYER_STATE_DRIVER)
    {
        new Float:x, Float:y, Float:z;
        GetPlayerPos(playerid, x, y, z);

}

BPS::CarRespawn(carid)
{

    {
