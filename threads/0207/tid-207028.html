        if(!strlen(tmp) || strval(tmp) < 1 || strval(tmp) > 53) return SendClientMessage(playerid, COLOR_YELLOW, "ERROR: Invalid Drift place. It can be only from 1 to 53!");
        PlayerGoesDrift(playerid, DriftCoords[strval(tmp)-1][0], DriftCoords[strval(tmp)-1][1], DriftCoords[strval(tmp)-1][2], DriftCoords[strval(tmp)-1][3], DriftCoords[strval(tmp)-1][4]);
        return 1;
    }
    return 0;
}
stock PlayerGoesDrift(playerid, Float:x, Float:y, Float:z, Float:va, Float:a)
{
    if(!IsPlayerInAnyVehicle(playerid))
    {
        SetPlayerPos(playerid, x, y, z);
        SetPlayerFacingAngle(playerid, a);
    }
    else
    {
        if(GetPlayerState(playerid) != PLAYER_STATE_DRIVER) return SendClientMessage(playerid, COLOR_YELLOW, "ERROR: You need to be the driver!");
        SetVehiclePos(GetPlayerVehicleID(playerid), x, y, z);
        SetVehicleZAngle(GetPlayerVehicleID(playerid), va);
    }
    GameTextForPlayer(playerid, "~w~Welcome to ~r~Drift ~w~!", 5000, 3);
    SetCameraBehindPlayer(playerid);
    return 1;
}
stock strtok(const string[], &index)
{
    while ((index < strlen(string)) && (string[index] <= ' ')) index++;
    new offset = index, result[20];
    while ((index < strlen(string)) && (string[index] > ' ') && ((index - offset) < (sizeof(result) - 1)))
    {
        result[index - offset] = string[index];
        index++;
    }
    result[index - offset] = EOS;
    return result;
}
