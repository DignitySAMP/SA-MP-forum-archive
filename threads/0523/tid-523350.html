pawn Code:
CMD:verplayers(playerid)
{
    for(new pt = 0; pt < MAX_PLAYERS; pt++)
    {
        if(IsPlayerAdmin(pt))
            SetPlayerMarkerForPlayer(playerid, pt, ( GetPlayerColor( 1 ) & 0xFFFFFF00 ) );
    }
    return 1;
}
