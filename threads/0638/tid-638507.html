stock GetVehicleModelID(modelname[])
{
new modelid; //Used for storing the modelid;     
for(new i; i < 212; i++)
{
    if(!strcmp(modelname,VehicleNames[i],true))   
    {      
        modelid = i + 400;
        return modelid;  
    }
}
return 0;
} 
And the command will be: (in case you use ZCMD)
PHP Code:
CMD:veh(playerid,params[])
{
    if(!IsPlayerAdmin(playerid)) return 1; //This means: "If the player is not rcon admin return 1"
    if(isnull(params))  return SendClientMessage(playerid,-1,"ERROR: No car"); //This means: "If the string is null"
    new c; //Used to store the modelid of car
        if(IsNumeric(params))  c = strval(params); //This means: "If params is numeric"
        else c = GetVehicleModelID(params);
    if(c<400 || c>611)   return SendClientMessage(playerid,-1,"ERROR: Invalid Car ID");
    new Float:pos[3],car;
    GetPlayerPos(playerid,pos[0],pos[1],pos[2]);
    car = CreateVehicle(c,pos[0],pos[1],pos[2],random(255),random(255),0,-1,1);
    PutPlayerInVehicle(playerid,car,0);
    SendClientMessage(playerid,-1,"SPAWNED CAR"); 
    return 1;
}
