These functions must be in a callback to work... in this case, OnFilterScriptInit()...

pawn Code:
public OnFilterScriptExit()
{
    print("\n--------------------------------------");
    print("DM Admin System");
    print("--------------------------------------\n");

    for(new i = 0; i < MAX_VEHICLES; i++)
    {
        if(AdminCar[i] == 1) DestroyVehicle(i);
    }

    for(new i = 0; i < MAX_PLAYERS; i++) TextDrawDestroy(fine[i]);
    for(new i = 0; i < MAX_PLAYERS; i++) TextDrawDestroy(warn[i]);
    for(new i = 0; i < MAX_PLAYERS; i++) TextDrawDestroy(spec[i]);
    for(new i = 0; i < MAX_PLAYERS; i++) TextDrawDestroy(ann);
    return 1;
}
