GetVehicleModelInfo(GetVehicleModel(gvi), VEHICLE_MODEL_INFO_WHEELSFRONT, w[0], w[1], w[2]);
GetVehicleModelInfo(GetVehicleModel(gvi), VEHICLE_MODEL_INFO_WHEELSREAR, b[0], b[1], b[2]);

f[0] = f[0] / 2.2;
w[2] = w[2] + 1.6;
b[2] = b[2] + 1.6;

AttachObjectToVehicle(Breaks[gvi][0], gvi, f[0], w[1], w[2], 180.0, 0.0, 0.0);
AttachObjectToVehicle(Breaks[gvi][1], gvi, -f[0], w[1], w[2], 180.0, 0.0, 0.0);

AttachObjectToVehicle(Breaks[gvi][2], gvi, f[0], b[1], b[2], 180.0, 0.0, 0.0);
AttachObjectToVehicle(Breaks[gvi][3], gvi, -f[0], b[1], b[2], 180.0, 0.0, 0.0);
