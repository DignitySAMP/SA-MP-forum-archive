// If there is no slot available for the cuffs object, 0 will be returned.

stock cuff_SetPlayerSkin(playerid, skin)
{
	new cuffs_id = GetPVarInt(playerid, "cuffs_object_index");
	if(cuffs_id != 0) // Cuffed, re-position cuffs object
	{
		SetPlayerAttachedObject(playerid, cuffs_id-1, 19418, 6, gCuffSkinData[skin][0], gCuffSkinData[skin][1], gCuffSkinData[skin][2], gCuffSkinData[skin][3], gCuffSkinData[skin][4], gCuffSkinData[skin][5], gCuffSkinData[skin][6], gCuffSkinData[skin][7], gCuffSkinData[skin][8]);
	}
	return SetPlayerSkin(playerid, skin);
}

	#define _ALS_SetPlayerSkin

#define SetPlayerSkin cuff_SetPlayerSkin

stock SetPlayerCuffed(playerid, cuffed=1)
{
	if(!IsPlayerConnected(playerid)) return -1;

	if(cuffed)
	{
		for(new i=MAX_PLAYER_ATTACHED_OBJECTS-1; i!=0; i--)
		{
			if(!IsPlayerAttachedObjectSlotUsed(playerid, i)) // Free slot
			{
				new skin = GetPlayerSkin(playerid);
				SetPVarInt(playerid, "cuffs_object_index", i+1);
				SetPlayerAttachedObject(playerid, i, 19418, 6, gCuffSkinData[skin][0], gCuffSkinData[skin][1], gCuffSkinData[skin][2], gCuffSkinData[skin][3], gCuffSkinData[skin][4], gCuffSkinData[skin][5], gCuffSkinData[skin][6], gCuffSkinData[skin][7], gCuffSkinData[skin][8]);
				SetPlayerSpecialAction(playerid, SPECIAL_ACTION_CUFFED);
				return 1;
			}
		}
		return 0; // No slot available
	}
	else
	{
		if(GetPVarInt(playerid, "cuffs_object_index") == 0) return 0; // No cuffs attached
		RemovePlayerAttachedObject(playerid, GetPVarInt(playerid, "cuffs_object_index")-1);
		SetPVarInt(playerid, "cuffs_object_index", 0);
		SetPlayerSpecialAction(playerid, 0);
		return 1;
	}
}
