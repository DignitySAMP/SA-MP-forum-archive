    DestroyPlayerObject(playerid, KillObject[playerid]);

    KillTimer(AllTimers[playerid][0]);
   KillTimer(AllTimers[playerid][1]);
   KillTimer(AllTimers[playerid][2]);
   StopAudioStreamForPlayer(playerid);
    if(IsValidPlayerObject(playerid, ObjectsSpawner[playerid][0]))

   }

   return 1;
}
