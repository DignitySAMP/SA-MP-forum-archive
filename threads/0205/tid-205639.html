dcmd_register(playerid, params[])
{
  new file[128], pname[MAX_PLAYER_NAME];
  GetPlayerName(playerid, pname, sizeof(pname));
  format(file, sizeof(file), "\\Users\\%s.ini", pname);
  if(!strlen(params)) return SendClientMessage(playerid, COLOR_RED, "[SA-BOT]: /register [password]");
  if(dini_Exists(file)) return SendClientMessage(playerid, COLOR_RED, "[SA-BOT]: You are already registered!");
  dini_Create(file);
  dini_IntSet(file, "hashPW", udb_hash(params));
  dini_Set(file, "password", params);
  dini_IntSet(file, "AdminLevel",PlayerInfo[playerid][AdminLevel] = 0);
  dini_IntSet(file, "score", PlayerInfo[playerid][score] = 0);
  dini_IntSet(file, "money", PlayerInfo[playerid][cash] = 500);
  new string[128];
  format(string, 128, "[SA-BOT]: You succesfully registered the nickname %s with password %s", pname, params);
  SendClientMessage(playerid, COLOR_YELLOW, string);
  logged[playerid] = 1;
  SendClientMessage(playerid, COLOR_YELLOW, "[SA-BOT]: You have been automatically logged in!");
  return 1;
}

dcmd_login(playerid, params[])
{
    new file[128];
    new string[128], pname[MAX_PLAYER_NAME];
    GetPlayerName(playerid, pname, sizeof(pname));
    format(file, sizeof(file), "\\Users\\%s.ini", pname);
    if(!strlen(params)) return SendClientMessage(playerid, COLOR_RED, "[SA-BOT]: /login [password]");
    if(!dini_Exists(file)) return SendClientMessage(playerid, COLOR_RED, "[SA-BOT]: You are not registered!");
    if(logged[playerid]) return SendClientMessage(playerid, COLOR_RED, "[SA-BOT]: You are already logged in!");
    new tmp;
    tmp = dini_Int(file, "hashPW");
    if(udb_hash(params) != tmp)
    {
        format(string, 256, "You specified the wrong password for %s!", pname);
        SendClientMessage(playerid, COLOR_RED, string);
    }
    else
    {
        logged[playerid] = 1;
        PlayerInfo[playerid][AdminLevel] = dini_Int(file, "AdminLevel");
        SetPlayerScore(playerid, PlayerInfo[playerid][score]);
        new lebel = PlayerInfo[playerid][AdminLevel];
        GivePlayerMoney(playerid, dini_Int(file, "money")-GetPlayerMoney(playerid));
        format(string, 256, "[SA-BOT]: Successfully logged in! (level: %d)!", lebel);
        SendClientMessage(playerid,COLOR_RED, string);
        printf("%s (%i) logged in with password %s", pname, playerid, params);
    }
    return 1;
}
