pawn Code:
SetPlayerTeamFromClass(playerid, classid) 
{ 
    if(classid >= 0 && classid <= 5 || classid == 0) gTeam[playerid] = COP;  // You add your newly added class to the check
    else if(classid == 6) gTeam[playerid] = FBI;
