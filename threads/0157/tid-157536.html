Code:
public OnPlayerCommandText(playerid, cmdtext[])
{
	new
	    tmp[20],
	    tmp2[128],
	    cmd[20],
	    id,
	    idx;
	    
	cmd = strtok(cmdtext, idx);
	
	if(!strcmp("/pm", cmd, true))
	{
	    tmp = strtok(cmdtext, idx);
	    tmp2 = strrest(cmdtext, idx);
		if(!strlen(tmp) || !strlen(tmp2))
		{
		    return SendClientMessage(playerid, 0xEE0000FF, "Uso: /pm [id] [mensaje]");
		}
		id = strval(tmp);
		if(!IsPlayerConnected(id))
		{
		    return SendClientMessage(playerid, 0xEE0000FF, "Jugador no conectado");
		}
		if(id != playerid) PM(id, playerid, tmp2);
		else SendClientMessage(playerid, 0xEE0000FF, "No puedes enviarte mensajes privados a ti mismo");
		return 1;
	}
	return 0;
}
Las funciones strtok y strrest:
Code:
stock strrest(const string[], &index)
{
	new length = strlen(string);
	while ((index < length) && (string[index] <= ' '))
	{
		index++;
	}
	new offset = index;
	new result[128];
	while ((index < length) && ((index - offset) < (sizeof(result) - 1)))
	{
		result[index - offset] = string[index];
		index++;
	}
	result[index - offset] = EOS;
	return result;
}

stock strtok(const string[], &index)
{
	new length = strlen(string);
	while ((index < length) && (string[index] <= ' '))
	{
		index++;
	}

	new offset = index;
	new result[20];
	while ((index < length) && (string[index] > ' ') && ((index - offset) < (sizeof(result) - 1)))
	{
		result[index - offset] = string[index];
		index++;
	}
	result[index - offset] = EOS;
	return result;
}
