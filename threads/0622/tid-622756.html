stock CountVehicles(filename[]) {
	new File:VehicleFile;
	new blank[256];
	new count = 0;
	if (fexist(V_FILE_SAVE)) {
		VehicleFile = fopen(V_FILE_SAVE);
		while(fread(VehicleFile, blank, sizeof blank)) {
			count++;
		}
		fclose(VehicleFile);
	}
	else {
		VehicleFile = fopen(filename);
		while(fread(VehicleFile, blank, sizeof blank)) {
			count++;
		}
		fclose(VehicleFile);
	}
	return count;
}
