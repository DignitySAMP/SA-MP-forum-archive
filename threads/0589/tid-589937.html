Решил в своём моде переработать систему проверки е-mail-а при регистрации.

PHP Code:
new F[40], S[20], T[4];
if(sscanf(text, "s[40]p<@>s[20]P<.>s[4]",F, S, T))print("true");else print("false");

PHP Code:

if(sscanf("Testmailmail.ru","s[40]p<@>s[20]P<.>s[4]",F, S, T))print("true");else print("false");
if(sscanf("Testmail@mailru","s[40]p<@>s[20]P<.>s[4]",F, S, T))print("true");else print("false");

if(sscanf("Testmail@.ru","s[40]p<@>s[20]P<.>s[4]",F, S, T))print("true");else print("false");

PHP Code:
ValidEmail(const mail[]){
    new c = strlen(mail);
    if(9>c>64)return 0;
    new F[32], S[20], T[4];
    if(sscanf(mail,"p<@>s[40]p<.>s[20]s[4]",F, S, T))return 0;
    else if(strlen(F) <4 || strlen(F) > 31 || strlen(S) <2 || strlen(S) >20 || strlen(T) <1 || strlen(T) >4)return 0;
    for(new i; i != c; i++)
    switch(mail[i]){
        case 'А'..'Я', 'а'..'я', ' ', 'ё', 'Ё', '№':return 0;
    }
    return 1;
}

PHP Code:
ValidEmail(const email[]) {
    new atPosition = strfind(email, "@"),
        len = strlen(email),
        dotPosition = strfind(email, ".", false, atPosition);
    if (atPosition == -1 || atPosition == 0 || atPosition == len - 1) {
        return 0;
    }
    if (dotPosition == -1 || dotPosition == 0 || dotPosition == len - 1 || dotPosition == len - 2 || atPosition + 1 == dotPosition) {
        return 0;
    }
    return (len > 5 ? 1 : 0);
}

function IsValidEmail(email)
{
    var at = email.indexOf("@");
    if(at < 1) return false;
    var dot = email.indexOf(".", at);
    if(dot == -1 || dot == (email.length - 1) || email[email.length - 1] == '.') return false;
