Make it for RCON. This is just plugin. Not FS
PHP Code:
public OnPlayerCommandText(playerid,cmdtext[])
{
    if(!strcmp("/playmusic",cmdtext,true))
    {
        if(IsPlayerAdmin(playerid))
        {
            Audio_SetPack("[your pack name, you will see once you download plugin]", true, true);
            for(new i=0;i<MAX_PLAYERS;i++)
            Audio_Play(i,audioid,false,true,false);
        }
        return 1;
    }
    return 0;
}
