SetPlayerTeamFromClass(playerid, classid)
{
  if(classid == 46 || classid == 47 || classid == 48 || classid == 49 || classid == 50 || classid == 51 || classid == 52 || classid == 53 || classid == 54)
  {
    PlayerInfo[playerid][pTeam] = TEAM_COP;
    PlayerInfo[playerid][pCopRadio] = 1;
  }
  if(classid == 55)
  {
    PlayerInfo[playerid][pTeam] = TEAM_SWAT;
  }
  if(classid == 43)
  {
    PlayerInfo[playerid][pTeam] = TEAM_FBI;
  }
  if(classid == 41)
  {
    PlayerInfo[playerid][pTeam] = TEAM_ARMY;
  }
  if(classid == 42)
  {
    PlayerInfo[playerid][pTeam] = TEAM_CIA;
  }
  if(classid == 59 || classid == 60 || classid == 61)
  {
    PlayerInfo[playerid][pTeam] = TEAM_MEDIC;
  }
  return 1;
}
