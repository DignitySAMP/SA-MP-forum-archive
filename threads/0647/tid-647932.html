Here is my old code of getting the keys (for that you need to hook the window instance of gta-sa(mp)):

Keyboard.h
Code:
#pragma once
namespace Keyboard {
	void WndProc(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam);
	void AddKey(BYTE byteType, BYTE byteKey);
};
Keyboard.cpp
Code:
#include "Keyboard.h"
//... includes like LiteRPC for memory hacking UDP connection of SAMP - instead use ur socket

bool CaptureKey[KEYBOARD_MAX_KEYS] = { 0 };

void Keyboard::WndProc(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam) // call this on WndProc of your GTA-SA(MP) instance
{
	switch (uMsg)
	{
		case WM_KEYUP:
		{
			if (wParam < 0 || wParam > KEYBOARD_MAX_KEYS)
				break;

			BYTE byteKey = static_cast<BYTE>(wParam);

			// get the char of pressed key
			DWORD dwProcessId = GetWindowThreadProcessId(hWnd, NULL);
			HKL dwhkl = GetKeyboardLayout(dwProcessId);
			BYTE vks[KEYBOARD_MAX_KEYS + 1];
			WORD w;
			GetKeyboardState(vks);
			int result = ToAsciiEx((unsigned short int)wParam, HC_ACTION, vks, &w, 0, dwhkl);

			if (CaptureKey[byteKey])
			{
				DEBUG_OUT("SENDING KEY %x with char %c\n", byteKey, (char)w);
				RPCS::Keyboard::SendKey(byteKey, w); // RPC sending key, you could use ur socket
			}

			break;
		}
	}
}

void Keyboard::AddKey(BYTE byteType, BYTE byteKey) // such a register to not spam the key
{
	if (byteKey < 0 || byteKey > KEYBOARD_MAX_KEYS)
		return;

	DEBUG_OUT("ADDING KEY %x\n", byteKey);

	switch (byteType)
	{
	case 0: // register
		CaptureKey[byteKey] = true;
		break;
	case 1: // unregister
		CaptureKey[byteKey] = false;
		break;
	}
}
