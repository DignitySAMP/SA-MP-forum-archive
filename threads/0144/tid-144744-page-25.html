stock Rank(Scor[], Top[], size, size2)
{
	new scortp;
	for(new tp = 0; tp < size2; tp++)
	{
	    scortp = 0;
	    for(new sp = 0; sp < size; sp++)
	    {
	        if(tp == 0)
	        {
	            if(Scor[sp] > scortp)
				{
					Top[tp] = sp;
					scortp = Scor[sp];
				}
			}
			else
			{
			    if(Scor[sp] > scortp)
			    {
			        if(Scor[sp] < Scor[Top[tp-1]])
			        {
			            Top[tp] = sp;
						scortp = Scor[sp];
			        }
			    }
			}
	    }
	}
	return 1;
}
