dcmd_setcash(playerid, params[])
{
    new PID, Cash;
    if(sscanf(params, "ui", PID, Cash)) return SendClientMessage(playerid, COLOR_RED, "USAGE: /SetCash < PlayerID > < Cash >");
    if(!IsPlayerConnected(PID)) return 0;

    SetPlayerMoney(PID, Cash);
