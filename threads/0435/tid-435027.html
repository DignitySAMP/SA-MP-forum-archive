stock CountConnectedPlayers()
{
	new PlayerCount = 0;
	for(new L = 0; L <= MAX_PLAYERS; L ++)
	{
        if(GetPlayerState(L) == PLAYER_STATE_NONE) continue;
		if(IsPlayerConnected(L)) PlayerCount++;
	}
	return PlayerCount;
}
