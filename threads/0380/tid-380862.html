public OnPlayerConnect(playerid)
{
	GetPlayerName(playerid, pName, 24);
	format(string, 128, "%s has joined the server!", pName);
	SCMTA(COL_GREEN, string);
	SendClientMessage(playerid, COL_BLUE, "Welcome to the {FF8040}server!{FF8040} Please read the {FF0000}/rules.{FF0000}");
	SendClientMessage(playerid, COL_BLUE, "If you need anything, feel free to ask in the chat, or use {FF0000}/help!{FF0000}");
	SetPlayerColor(playerid, 0xFFFFFFFF);
	return 1;
}

public OnPlayerDisconnect(playerid, reason)
{
	new string3[64];
	new pName2[24];
    GetPlayerName(playerid, pName2, 24);
    switch(reason)
    {
        case 0: format(string3, 128,"%s left the server. (Timeout)", pName2);
        case 1: format(string3, 128,"%s left the server. (Leaving)", pName2);
        case 2: format(string3, 128,"%s left the server. (Kicked/Banned)", pName2);
    }
	SCMTA(COL_GREEN, string);
	return 1;
}

public OnPlayerDeath(playerid, killerid, reason)
{
	SendDeathMessage(killerid, playerid, reason);
	new pName3[24];
	new string3[128];
	GetPlayerName(playerid, pName3, 24);
	format(string3, 128, "%s has been killed!", pName3);
	SCMTA(COL_YELLOW, string3);
	GameTextForPlayer(playerid, "~w~hahaha you died!", 4000, 2);
	GivePlayerMoney(playerid, -500);
	ClearAnimations(playerid);
	return 1;
}

public OnPlayerSpawn(playerid)
{
	GivePlayerWeapon(playerid, 5, 999);
	GivePlayerWeapon(playerid, 24, 9999);
	GivePlayerWeapon(playerid, 27, 9999);
	GivePlayerWeapon(playerid, 29, 9999);
	SetPlayerColor(playerid, PlayerColors[playerid]);
	return 1;
}

public OnPlayerText(playerid, text[])
{
	SetPlayerChatBubble(playerid, text, 0xFF0000FF, 100.0, 10000);
	return 1;
}

public OnGameModeExit()
{
	print("Zacklogan and NoahF's awesome gamemode is exiting.. :(");
	return 1;
}
