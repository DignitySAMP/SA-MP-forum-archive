BubbleSort( Array[], iSize) // bubblesort (by drakins)
{
    new
        iLoop       = 0x00000000,
        iCircuito   = 0x00000000;

    while(++iLoop < iSize)
    {
        iCircuito = 0xFFFFFFFF;
        while(++iCircuito != iLoop) b_swap(Array[iLoop],Array[iCircuito]);
    }
    return Array;
}
