Медленно. Вот функция быстрой сортировки с википедии, переведённая с C в Pawn.
pawn Code:
stock qSort(array[], low, high)
{
    new i = low,
        j = high,
        x = array[(low + high) / 2];

    do {
        while (array[i] < x) {
            i++;
        }
        while (array[j] > x) {
            j--;
        }
        
        if (i <= j) {
            new temp = array[i];
            array[i] = array[j];
            array[j] = temp;
            
            i++;
            j--;
        }
    } while (i < j);
    
    if (low < j) {
        qSort(array, low, j);
    }
    if (i < high) {
        qSort(array, i, high);
    }
}
