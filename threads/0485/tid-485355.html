CMD:buildrace(playerid, params[])
{
if(!IsPlayerAdmin(playerid)) return SendClientMessage(playerid, RED, "<!> Usted no es un administrador!");
if(BuildRace != 0) return SendClientMessage(playerid, RED, "<!> Ya hay alguien que construye una carrera!");
if(RaceBusy == 0x01) return SendClientMessage(playerid, RED, "<!> Espere hasta que termine la primera carrera!");
if(IsPlayerInAnyVehicle(playerid)) return SendClientMessage(playerid, RED, "<!> Por favor, deje su vehнculo primero!");
BuildRace = playerid+1;
ShowDialog(playerid, 599);
return 1;
}
//*=============================================================================
CMD:startrace(playerid, params[])
{
if(!IsPlayerAdmin(playerid)) return SendClientMessage(playerid, RED, "<!> Usted no es un administrador!");
if(BuildRace != 0) return SendClientMessage(playerid, RED, "<!> Hay una carrera iniciada, espere que termine!");
if(RaceBusy == 0x01 || RaceStarted == 1) return SendClientMessage(playerid, RED, "<!> Hay una Carrera en la actualidad, espere a que finalize!");
if(isnull(params)) return SendClientMessage(playerid, RED, "<!> /startrace [racename]");
LoadRace(playerid, params);
return 1;
}
//*=============================================================================
CMD:stoprace(playerid, params[])
{
if(!IsPlayerAdmin(playerid)) return SendClientMessage(playerid, RED, "<!> Usted no es un administrador!");
if(RaceBusy == 0x00 || RaceStarted == 0) return SendClientMessage(playerid, RED, "<!> No hay carrera para detener!");
SendClientMessageToAll(RED, ">>Un administrador detuvo la carrera actual!");
return StopRace();
}
//*=============================================================================
CMD:join(playerid, params[])
{
if(RaceStarted == 1) return SendClientMessage(playerid, RED, "<!> La carrera ya ha comenzado! Espere hasta que termine!");
if(RaceBusy == 0x00) return SendClientMessage(playerid, RED, "<!> No hay carrera para unirse");
if(Joined[playerid] == true) return SendClientMessage(playerid, RED, "<!> Ya se uniу a una carrera!");
if(IsPlayerInAnyVehicle(playerid)) return SetTimerEx("SetupRaceForPlayer", 2500, 0, "e", playerid), RemovePlayerFromVehicle(playerid), Joined[playerid] = true;
SetupRaceForPlayer(playerid);
Joined[playerid] = true;
return 1;
}

public OnPlayerEnterRaceCheckpoint(playerid)
{
if(CPProgess[playerid] == TotalCP -1)
{
new
TimeStamp,
TotalRaceTime,
string[128],
pName[MAX_PLAYER_NAME],
rTime[3],
Prize[2]
;
Position++;
FinishCount++;
GetPlayerName(playerid, pName, sizeof(pName));
TimeStamp = GetTickCount();
TotalRaceTime = TimeStamp - RaceTick;
ConvertTime(var, TotalRaceTime, rTime[0], rTime[1], rTime[2]);
switch(Position)
{
case 1: Prize[0] = (random(random(5000)) + 10000), Prize[1] = 10;
case 2: Prize[0] = (random(random(4500)) + 9000), Prize[1] = 9;
case 3: Prize[0] = (random(random(4000)) + 8000), Prize[1] = 8;
case 4: Prize[0] = (random(random(3500)) + 7000), Prize[1] = 7;
case 5: Prize[0] = (random(random(3000)) + 6000), Prize[1] = 6;
case 6: Prize[0] = (random(random(2500)) + 5000), Prize[1] = 5;
case 7: Prize[0] = (random(random(2000)) + 4000), Prize[1] = 4;
case 8: Prize[0] = (random(random(1500)) + 3000), Prize[1] = 3;
case 9: Prize[0] = (random(random(1000)) + 2000), Prize[1] = 2;
default: Prize[0] = random(random(1000)), Prize[1] = 1;
}
format(string, sizeof(string), ">> \"%s\" ha terminado la carrera en la posiciуn \"%d\".", pName, Position);
SendClientMessageToAll(WHITE, string);
format(string, sizeof(string), "    - Time: \"%d:%d.%d\".", rTime[0], rTime[1], rTime[2]);
SendClientMessageToAll(WHITE, string);
format(string, sizeof(string), "    - Premio: \"$%d \".", Prize[0], Prize[1]);
SendClientMessageToAll(WHITE, string);
GivePlayerMoney(playerid, Prize[0]);
SetPlayerScore(playerid, GetPlayerScore(playerid) + Prize[1]);
DisablePlayerRaceCheckpoint(playerid);
CPProgess[playerid]++;
if(FinishCount >= JoinCount) return StopRace();
}
else
{
CPProgess[playerid]++;
CPCoords[3][CPProgess[playerid]]++;
RacePosition[playerid] = floatround(CPCoords[3][CPProgess[playerid]], floatround_floor);
SetCP(playerid, CPProgess[playerid], CPProgess[playerid]+1, TotalCP, RaceType);
PlayerPlaySound(playerid, 1137, 0.0, 0.0, 0.0);
}
return 1;
}

{

//*=============================================================================
function StartRace()
{
ForEach(i, MAX_PLAYERS)
{
if(Joined[i] == true)
{
TogglePlayerControllable(i, true);
PlayerPlaySound(i, 1057, 0.0, 0.0, 0.0);
GameTextForPlayer(i, "~g~GO GO GO", 2000, 5);
SetCameraBehindPlayer(i);
}
}
rCounter = SetTimer("RaceCounter", 900, 1);
RaceTick = GetTickCount();
RaceStarted = 1;
KillTimer(CountTimer);
return 1;
}
//*=============================================================================
function StopRace()
{
KillTimer(rCounter);
RaceStarted = 0;
RaceTick = 0;
RaceBusy = 0x00;
JoinCount = 0;
FinishCount = 0;
ForEach(i, MAX_PLAYERS)
{
if(Joined[i] == true)
{
DisablePlayerRaceCheckpoint(i);
DestroyVehicle(CreatedRaceVeh[i]);
Joined[i] = false;
TextDrawHideForPlayer(i, RaceInfo[i]);
CPProgess[i] = 0;
KillTimer(InfoTimer[i]);
}
}
SendClientMessageToAll(GREEN, ">> Race time is over!");
return 1;
}
//*=============================================================================
function RaceCounter()
{
if(RaceStarted == 1) RaceTime--;
if(RaceTime <= 0)
{
StopRace();
}
return 1;
}
