GetVipLevel(playerid)
{
    new str[10];
    switch(PlayerInfo[playerid][pVIP])
    {
        case 0: str = "None";
        case 1: str = "Bronze";
        case 2: str = "Silver";
        case 3: str = "Gold";
    }
    return str;
}
GetVipColor(playerid)
{
    new str[10];
    switch(PlayerInfo[playerid][pVIP])
    {
        case 1: str = "{996600}";
        case 2: str = "{adad85}";
        case 3: str = "{e68a00}";
    }
    return str;
}
