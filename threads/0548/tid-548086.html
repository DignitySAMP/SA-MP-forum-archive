Code:
stock IsValidObjectModelID(ModelID)
{
	if((ModelID >= 321 && ModelID <= 326) || (ModelID >= 330 && ModelID <= 331) || (ModelID >= 333 && ModelID <= 339) || (ModelID >= 341 && ModelID <= 344) || (ModelID >= 346 && ModelID <= 363) || (ModelID >= 365 && ModelID <= 372) || (ModelID >= 1433 && ModelID <= 13594) || (ModelID >= 5482 && ModelID <= 5512) || (ModelID >= 966 && ModelID <= 998) || (ModelID >= 1210 && ModelID <= 1325) || (ModelID >= 1420 && ModelID <= 1620) || (ModelID >= 1971 && ModelID <= 4522)) return 1;
	return 0;
}
