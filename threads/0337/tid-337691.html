pawn Code:
// Turn the angle to point right (-90 degrees)
angle = ((angle - 90.0) < 0.0) ? (360.0 + (angle - 90.0)) : (angle - 90.0);

// Move the coordinades towards the angle
x += (distance * floatsin(-angle, degrees));
y += (distance * floatcos(-angle, degrees));
