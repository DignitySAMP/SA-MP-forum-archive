		if(isCuffed(otherID))

				format(msg, sizeof(msg), CHAT_YELLOW"%s"CHAT_WHITE" has been arrested by "CHAT_YELLOW"%s"CHAT_WHITE" for"CHAT_YELLOW" %d"CHAT_WHITE" seconds.", GetName(otherID), GetName(playerid), time);

				Info(playerid, msg);
