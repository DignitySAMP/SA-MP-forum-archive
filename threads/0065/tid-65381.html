Quick search and i found this
pawn Code:
aLog(file[], msg[])
{
  new fName[256], entry[256], File: hFile;
  format(fName, sizeof(fName), "\\Logs\\%s", file);
  if(!fexist(fName))
  {
    dini_Create(fName);
  }
  format(entry, sizeof(entry), "%s\r\n", msg);
  hFile = fopen(fName, io_append);
  fwrite(hFile, entry);
  fclose(hFile);
}
Paste that somewhere in your script and you should be okay
