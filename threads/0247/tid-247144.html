esse abaixo usa os comando /entrarfila = usado pelos player para entrar na fila de espera
/sairfila = se o player tiver usado /entrarfila e nao precisa mais de atendimento
/fila = para os adms ver quem estб na fila
/afila = usando assim: /afila [playerid/parte-do-nick] para o adm ir a alguem q entrou na fila, Script editado para o GF e derivados ^^


Code:
	if(strcmp(cmd, "/entrarfila", true) == 0)
	{
	    if(IsPlayerConnected(playerid))
	    {
	        if(PlayerInfo[playerid][pAge] == 0){
                PlayerInfo[playerid][pAge] = 1;
                SendClientMessage(playerid, COLOR_GREY, "Vocк entrou na fila de atendimento. Aguarde um Admin atender.");
	        }else{
	            SendClientMessage(playerid, COLOR_GREY, "Vocк jб estб na fila de atendimento. Para sair digite: /sairfila");
	            }
	    }
 	}
	if(strcmp(cmd, "/sairfila", true) == 0)
	{
	    if(IsPlayerConnected(playerid))
	    {
	        if(PlayerInfo[playerid][pAge] == 0){
                SendClientMessage(playerid, COLOR_GREY, "Vocк nгo estб na fila de atendimento. Para entrar digite: /entrarfila");
	        }else{
	       		PlayerInfo[playerid][pAge] = 0;
	            SendClientMessage(playerid, COLOR_GREY, "Vocк saiu da fila de atendimento.");
	            }
	    }
 	}
	if (strcmp(cmd, "/fila", true) == 0)
	{
		if(IsPlayerConnected(playerid))
		{
			SendClientMessage(playerid, COLOR_YELLOW, "|-Fila de Atendimento-|");
			for(new i = 0; i < MAX_PLAYERS; i++)
			{
				if(IsPlayerConnected(i))
				{
					if(PlayerInfo[i][pAge] == 1){GetPlayerName(i, sendername, sizeof(sendername));format(string, 256, "%s (%d)", sendername,i);SendClientMessage(playerid, COLOR_YELLOW, string);}
				}
			}
		}
		return 1;
	}

    if (strcmp(cmd, "/afila", true) == 0)
	{
		if(IsPlayerConnected(playerid))
		{

		    tmp = strtok(cmdtext, idx);
			if(!strlen(tmp))
			{
				SendClientMessage(playerid, COLOR_GRAD2, "USO: /afila [playerid/Parte-do-Nick]");
				return 1;
			}
			new Float:plocx,Float:plocy,Float:plocz;
			new plo;
			plo = ReturnUser(tmp);
			if (IsPlayerConnected(plo))
			{
			    if(plo != INVALID_PLAYER_ID)
			    {
					new nome[24];
                    GetPlayerName(playerid, nome, sizeof nome);
                    if(PlayerInfo[playerid][pAdmin] >= 1 || strfind(nome, ADMPLAYER, true) == 0 || strfind(nome, ADMPLAYER2, true) == 0 )
					{
					    if(PlayerInfo[plo][pAge] == 1){
							if(Spectate[playerid] != 255)
							{
								Spectate[playerid] = 256;
							}
							GetPlayerPos(plo, plocx, plocy, plocz);
							if(PlayerInfo[plo][pInt] > 0)
							{
								SetPlayerInterior(playerid,PlayerInfo[plo][pInt]);
								PlayerInfo[playerid][pInt] = PlayerInfo[plo][pInt];
								PlayerInfo[playerid][pLocal] = PlayerInfo[plo][pLocal];
							}
							if(PlayerInfo[playerid][pInt] == 0)
							{
								SetPlayerInterior(playerid,0);
							}
							if(plocz > 530.0 && PlayerInfo[plo][pInt] == 0) //the highest land point in sa = 526.8
							{
								SetPlayerInterior(playerid,1);
								PlayerInfo[playerid][pInt] = 1;
							}
							if (GetPlayerState(playerid) == 2)
							{
								new tmpcar = GetPlayerVehicleID(playerid);
								SetVehiclePos(tmpcar, plocx, plocy+4, plocz);
								TelePos[playerid][0] = 0.0;TelePos[playerid][1] = 0.0;
							}
							else
							{
								SetPlayerPos(playerid,plocx,plocy+2, plocz);
							}
							GetPlayerName(plo, giveplayer,256);
							GetPlayerName(playerid, sendername,256);
		               		format(string, sizeof(string), "[Info] Vocк foi atender ao chamado de %s.", giveplayer);
		               	 	SendClientMessage(playerid, COLOR_WHITE, string);
		               	 	format(string, sizeof(string), "[Info] O admin %s, veio para te atender.", sendername);
		               	 	SendClientMessage(plo, COLOR_WHITE, string);
		                	new string2[256];
		                	format(string2, sizeof(string), "[FILA] O Admin %s foi atender o player %s.", giveplayer,sendername);
							ABroadCast(COLOR_KICKBAN,string2,1);
							PlayerInfo[plo][pAge] = 0;
	   					}else{
	   						SendClientMessage(playerid, COLOR_GRAD1, "O player nгo estб na fila.");
   						}
					}
					else
					{
						SendClientMessage(playerid, COLOR_GRAD1, "Vocк nгo estб autorizado a usar este comando.");
					}
				}
			}
			else
			{
				format(string, sizeof(string), "   %d nгo й um player ativo.", plo);
				SendClientMessage(playerid, COLOR_GRAD1, string);
			}

		}
		return 1;
	}
