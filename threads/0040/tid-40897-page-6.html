#define getarray(%1,%2) \
    do \
    { \
        static _c = false; \
        for(_c = 0; %2[_c] != 0;) %2[_c++] = false; \
        for(_c = 0; getarg(%1, _c) != 0;) %2[_c] = getarg(%1, _c++); \
