pawn Code:
if(!strcmp(cmdtext, "/buildingjump", true))
        {
        if(IsPlayerInAnyVehicle(playerid))
        {
            SetVehiclePos(GetPlayerVehicleID(playerid), 3335.7023,809.5499,8.0119);
            SetVehicleZAngle(GetPlayerVehicleID(playerid), 0.0);
            LinkVehicleToInterior(GetPlayerVehicleID(playerid), 0);
        }
        else
        {
            SetPlayerPos(playerid, 3335.7023,809.5499,8.0119);
            SetPlayerFacingAngle(playerid, 0.0);
        }
        SetPlayerInterior(playerid, 0);
        SendClientMessage(playerid, COLOR_LIGHTBLUE, " Welcome to the Building Jump!");
        return 1;
    }
