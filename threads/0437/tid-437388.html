public OnPlayerCommandText(playerid, cmdtext[])
{

    {
        new carid = GetPlayerVehicleID(playerid); 
        if(!IsTrailerAttachedToVehicle(carid)) return SendClientMessage(playerid, COLOR_RED, "You need to attach a trailer!!!");

        {
            ToysJob[playerid] = 1;
