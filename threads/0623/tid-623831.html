stock converteNumber(n)
{
    new dia, hr, mn, seg, resto;

    resto = n;
    seg = resto % 60;
    resto /= 60;
    mn = resto % 60;
    resto /= 60;
    hr = resto % 24;
    resto /= 24;
    dia = resto;

    new str[50];
    format(str, sizeof(str), "%ddias, %02dh %02dm %02ds", dia, hr, mn, seg);
    return str;
}
