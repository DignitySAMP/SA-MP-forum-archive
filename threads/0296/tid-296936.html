pawn Code:
#include <a_samp>
#include <YSI\y_ini>

new
    Float: gLastX[ MAX_PLAYERS ],
    Float: gLastY[ MAX_PLAYERS ],
    Float: gLastZ[ MAX_PLAYERS ]
;

forward @load_user_position( playerid, name[], value[] );

@load_user_position( playerid, name[], value[] )
{
    INI_Float( "LastX", gLastX[ playerid ] );
    INI_Float( "LastY", gLastY[ playerid ] );
    INI_Float( "LastZ", gLastZ[ playerid ] );
    
    return 1;
}

stock user_position_path( playerid )
{
    new
        uName[ MAX_PLAYER_NAME ]
    ;
        
    GetPlayerName( playerid, uName, MAX_PLAYER_NAME );
    format( _STR, sizeof ( _STR ), "%s.ini", uName );
    return
        _STR;
}

public OnPlayerConnect( playerid )
{
    INI_ParseFile( user_position_path ( playerid ), "load_user_%s", .bExtra = true, .extra = playerid );
    
    return 1;
}

public OnPlayerSpawn( playerid )
{
    SetPlayerPos( playerid, gLastX[ playerid ], gLastY[ playerid ], gLastX[ playerid ] );
    
    return 1;
}

public OnPlayerDisconnect( playerid, reason )
{
    new
        INI:File = INI_Open( user_position_path ( playerid ) ), _P[ 3 ]
    ;
    GetPlayerPos( playerid, _P[ 0 ], _P[ 1 ], _P[ 2 ] );
    
    INI_SetTag(File,"position");
    INI_WriteFloat(File,"LastX", _P[ 0 ] );
    INI_WriteFloat(File,"LastY", _P[ 1 ] );
    INI_WriteFloat(File,"LastZ", _P[ 2 ] );
    INI_Close(File);
    
    return 1;
}
