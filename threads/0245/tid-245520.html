//above:
new Gate1, Gate2, Gate3;

public OnPlayerCommandText(playerid,cmdtext[]) 
{
    dcmd(aopen,5,cmdtext);
    dcmd(aclose,6,cmdtext);
    return 0;
}


dcmd_aopen(playerid,params[])//Open
{
    #pragma unused params
    if (PlayerInfo[playerid][level] == 0 && IsPlayerAdmin(playerid) == 0) return SendClientMessage(playerid,COLOR_RED,"You need to be at least a vip to use this cmd !");
//^ ^This is my admin system but you can change it to yours...^ ^
    MoveObject(Gate1, -2888.33471680,441.76980591,4.00218201, 5);

    MoveObject(Gate3, -2888.33471680,441.76980591,4.00218201, 5);
    SendClientMessage(playerid, COLOR_ORANGE, "You have opened the gates to the admin island ! Use /aclose to close them !");
    return 1;
}
dcmd_aclose(playerid,params[])//Close
{
    #pragma unused params
    if (PlayerInfo[playerid][level] == 0 && IsPlayerAdmin(playerid) == 0) return SendClientMessage(playerid, COLOR_RED, "You need to be at least a vip to use this cmd !");
