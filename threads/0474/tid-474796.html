    {
         TogglePlayerDynamicCP(playerid, rota2, 0); // desliga a rota 2
        TogglePlayerDynamicCP(playerid, rota3, 1); // liga a rota 3
        return 0;
    }
    return 1;
}
public CarregarRotas(playerid)
{
    rota1 = CreateDynamicCP(x, y, z, 4, -1, -1, playerid, 3000.0);
