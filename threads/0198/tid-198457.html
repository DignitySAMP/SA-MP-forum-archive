public OnPlayerCommandText(playerid, cmdtext[])
{
    if (strcmp("a", cmdtext, true, 5) == 0)
    {
        SetPlayerPos(playerid ,0,0,0);
        return 1;
    }
    if (strcmp("/lssh", cmdtext, true, 5) == 0)
    {
        SetPlayerPos(playerid ,3479.9443,-687.4036,1.8531);
        return 1;
    }
    if (strcmp("/heal", cmdtext, true, 5) == 0)
    {
        SetPlayerHealth(playerid ,100);
        return 1;
    }

    if (strcmp("/kill", cmdtext, true, 5) == 0)
    {
        SetPlayerHealth(playerid ,0);
        return 1;
    }
    if (strcmp("/help", cmdtext, true, 5) == 0)
    {
        SendClientMessage(playerid, COLOR_YELLOW, "heal");
        SendClientMessage(playerid, COLOR_YELLOW, "kill");
        SendClientMessage(playerid, COLOR_YELLOW, "More commands will be added soon.");
        return 1;
    }
    return 0;
}
