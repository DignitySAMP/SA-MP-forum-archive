Try to use that example:
Code:
#define DIALOG_ONE 0
#define DIALOG_TWO 1

public OnDialogResponse(playerid,dialogid,response,listitem,inputtext[])
{
	switch(dialogid)
	{
	    case DIALOG_ONE:
	    {
	        if(response)
	        {
	            ShowPlayerDialog(playerid,DIALOG_TWO,DIALGO_STYLE_MSGBOX,"Caption","Info","Back","Close");
	        }
	    }
	    case DIALOG_TWO:
	    {
	        if(!response)
	        {
	            ShowPlayerDialog(playerid,DIALOG_ONE,DIALGO_STYLE_MSGBOX,"Caption","Info","Next","Close");
	        }
	    }
	}
	return 1;
}

COMMAND:test(playerid,params[])
{
	ShowPlayerDialog(playerid,DIALOG_ONE,DIALGO_STYLE_MSGBOX,"Caption","Info","Next","Close");
	return 1;
}
