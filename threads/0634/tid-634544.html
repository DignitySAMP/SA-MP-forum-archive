CMD:buygun(playerid, params[])
{
     new weapon[128]; 
     if(sscanf(params, "s[128]", weapon))
            return SendClientMessage(playerid, -1, "USAGE: /buygun [ Desert Eagle ]");
    if(strcmp(weapon, "deserteagle", true) == 0)
