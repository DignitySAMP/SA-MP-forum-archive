CMD:gotocar(playerid, params[])
{
    new vehid;
    if(!sscanf(params, "d", vehid))
    {
        new Float:Xv, Float:Yv, Float:Zv;
        GetVehiclePos(vehid, Xv, Yv, Zv);
        SetPlayerPos(playerid, Xv+1, Yv+1, Zv+1);
        return 1;
    }
    else return SendClientMessage(playerid, -1, "USAGE: /gotocar [vehicle ID]");
