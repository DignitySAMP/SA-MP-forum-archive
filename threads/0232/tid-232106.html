stock GetCompID(compname[]) // Modified by [L3th4l]
{
    for(new i = 0; i < sizeof(ComponentNames); ++i)
    {
        if(strfind(ComponentNames[i], compname, true) != -1)
        {
            return i + 1000;
        }
    }
    return -1;
}
Usage:
pawn Code:
printf("Component ID: %i", GetCompID("Nitro: 10 Times"));
Would return: "Component ID: 1010"
