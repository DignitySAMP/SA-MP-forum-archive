public OnPlayerDeath(playerid, killerid, reason)
{
    if(killerid == INVALID_PLAYER_ID)
    {
        ResetPlayerMoney(playerid);
    }
    else
    {
        playercash = GetPlayerMoney(playerid);
        if(playercash > 0)
        {
            GivePlayerMoney(killerid, playercash);
            ResetPlayerMoney(playerid);
        }
    }
    return 1;
}
