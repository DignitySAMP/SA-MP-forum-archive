public SendAdminMessage(color,const string[],level)
{
    for(new i = 0; i < MAX_PLAYERS; i++)
    {
        if(IsPlayerConnected(i))

            if (PlayerInfo[i][pAdmin] >= level)
            {

            }
        }
    }
    return 1;
}
