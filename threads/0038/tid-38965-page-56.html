#define INI_WriteString(%0,%1,%2) fwrite(%0, %1), fputchar(%0, '='), fwrite(%0, %2), fwrite(%0, "\r\n")
#define INI_Int(%0,%1,%2)         do{new _string[128];while(fread(%0,_string)){if(strcmp(_string,%1,false,strlen(%1))==0&&_string[strlen(%1)]=='='){strmid(_string,_string,strlen(%1)+1,strlen(_string)-2,128);%2=strval(_string);fseek(%0,0,seek_start);break;}}}while(FALSE)
#define INI_Float(%0,%1,%2)       do{new _string[128];while(fread(%0,_string)){if(strcmp(_string,%1,false,strlen(%1))==0&&_string[strlen(%1)]=='='){strmid(_string,_string,strlen(%1)+1,strlen(_string)-2,128);%2=floatstr(_string);fseek(%0,0,seek_start);break;}}}while(FALSE)
#define INI_String(%0,%1,%2,%3)   do{new _string[128];while(fread(%0,_string)){if(strcmp(_string,%1,false,strlen(%1))==0&&_string[strlen(%1)]=='='){strmid(%2,_string,strlen(%1)+1,strlen(_string)-2,%3);fseek(%0,0,seek_start);break;}}}while(FALSE)
#define INI_Close(%0)             fclose(%0)
