What about this?

TogglePlayerSpectating
SetSpawnInfo
SpawnPlayer

pawn Code:
new FirstConnect[MAX_PLAYERS] = 0; // top of your script somewhere under includes

public OnPlayerConnect(playerid)
{
    TogglePlayerSpectating(playerid, 1);
    FirstConnect[playerid] = 1;
    SetSpawnInfo(playerid, team, skin, x, y, z, angle, weap1, weap1_ammo, weap2, weap2_ammo, weap3, weap3_ammo);
    SpawnPlayer(playerid);
    return 1;
}

public OnPlayerSpawn(playerid)
{
    if(FirstConnect[playerid] == 1)
    {
        TogglePlayerSpectating(playerid, 0);
        FirstConnect[playerid] = 0;
    }
    return 1;
}
Notice: You can't directly copy paste my code you'll get errors. You must change the SetSpawnInfo values.
