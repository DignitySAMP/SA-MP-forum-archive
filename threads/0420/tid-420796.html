pawn Code:
forward KickPublic(playerid);
public KickPublic(playerid) { Kick(playerid); }

KickWithMessage(playerid, message[])
{
    SendClientMessage(playerid, COLOR_RED, message); // define this color at the top of your script
    SetTimerEx("KickPublic", 1000, 0, "d", playerid);
}
pawn Code:
KickMessage(id,str); // put this in your kick or ban command
// Replace 'id' with the person being kicked and 'str' with whatever you named the string
