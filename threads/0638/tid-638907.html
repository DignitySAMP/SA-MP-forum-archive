stock aNombre(playerid)
{
	new pname[MAX_PLAYER_NAME], i_pos;
	GetPlayerName(playerid, pname, sizeof(pname));
	while((i_pos = strfind(pname, "_", false, i_pos)) != -1) pname[i_pos] = ' ';
	return pname;
}
