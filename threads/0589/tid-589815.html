Code:
GetNextVirtualWorld()
{
    for(new i = 0; i < MAX_PLAYERS; i++)
    {
        if(!IsPlayerConnected(i))
            continue;

        if(GetPlayerVirtualWorld(i) == i)
            continue;

        return i;
    }

    return -1;
}

public OnPlayerConnect(playerid)
{
    new vID = GetNextVirtualWorld();

    if(vID == -1)
        return 1;

    SetPlayerVirtualWorld(playerid, vID);
}
