  5. Никак. Хотя теоретически можно попробовать их удалить RemoveBuildingForPlayer, правда есть слухи что если удалить "скриптованые" игрой объекты появляется вероятность "падения" клиента.

"отключить" стандартные гаражи тюнинга и перекрасок на данный момент,
можно только создав перед ними альтернативные гаражные двери (CreateObject) по другому никак.

pawn Code:
// альтернативные ворота для всех тюнинг гаражей
lstg1 = CreateObject(3095, 1043.18, -1026.06, 29.72,   0.00, 90.00, 90.11);
lstg2 = CreateObject(3095, 2644.51, -2039.29, 11.79,   0.00, 90.00, 270.00);
sftg1 = CreateObject(3095, -1935.62, 238.49, 33.30,   0.00, 90.00, 90.00);
sftg2 = CreateObject(3095, -2715.97, 217.39, 3.24,   0.00, 90.00, 180.00);
lvtg1 = CreateObject(3095, 2386.81, 1043.34, 9.81,   0.00, 90.00, 90.00);
