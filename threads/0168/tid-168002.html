OnFilterScriptInit:

pawn Code:
SetTimer("AutoUpdateBegin", 5000, 1);
Some where down:

pawn Code:
forward AutoUpdateBegin();
public AutoUpdateBegin()
{
    for(new i=0;i<MAX_PLAYERS;i++)
    {
        if(IsPlayerConnected(i))
        {
            if(AccInfo[i][LoggedIn] == 1)
            {
                if(AccInfo[i][LoggedIn] == 1)
                SavePlayerStats(i);
            }
        }
    }
    return 1;
}
