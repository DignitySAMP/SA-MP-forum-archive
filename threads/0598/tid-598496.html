CMD:removervip(playerid,params[])
{

    {
        new ID;

            SendClientMessage(playerid,0xBFBFBFAA,"Use: /removervip ID.");
            return 1;
        }
        if(!IsPlayerConnected(ID))
        {
            SendClientMessage(playerid,0xBFBFBFAA,"Player nгo online.");

        }

        {
            if(VIP[ID] == 0)
            {

            else
            {
                GetPlayerName(ID,Nomev,sizeof(Nomev));
