            {
                idx++;
            }
            new offset = idx;
            new result[128];

            {
                result[idx - offset] = cmdtext[idx];
                idx++;
            }
            result[idx - offset] = EOS;
            if(!strlen(result))
            {
                SendClientMessage(playerid, WHITE, "SYNTAX: /me [action]");
                return 1;
            }
