stock GetVehiclePosEx(veh,&Float:x,&Float:y,&Float:z,Float:dist,Float:Zangle)
{
GetVehiclePos(veh,x,y,z);
new Float:r;
GetVehicleZAngle(veh,r);
Zangle=-(r+Zangle);
x = x + (dist * floatsin(Zangle,degrees));
y = y + (dist * floatcos(Zangle,degrees));
}
