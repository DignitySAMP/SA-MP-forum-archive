        {
            SetPlayerVirtualWorld(playerid, cellmax-vehicleid);

            SetCameraBehindPlayer(playerid);
        }
    }
    for(new i;i<sizeof(VehicleUniversity);i++)
    {
        if(IsPlayerInVehicle(playerid,VehicleUniversity[i]))
        {
            if(PlayerInfo[playerid][pFac] < 13 || PlayerInfo[playerid][pFac] > 15)

                break;
            }
        }
    }
    return 1;
}
