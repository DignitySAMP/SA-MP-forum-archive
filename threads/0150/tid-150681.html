If you found the file it should look something like this:
Code:
[boot loader]
timeout=30
default=multi(0)disk(0)rdisk(0)partition(1)\WINDOWS
[operating systems]
multi(0)disk(1)rdisk(0)partition(1)\WINDOWS="Windows 7 Professional" /fastdetect   <-- YOUR WINDOWS
multi(0)disk(0)rdisk(0)partition(2)\WINNT="Windows XP Professional" /fastdetect    <-- THE WINDOWS YOU WANT TO DELETE
