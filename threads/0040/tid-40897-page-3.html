This is not the spot, but here you go:

pawn Code:
if(!strcmp(cmdtext,"/c",true,2))
{
  if(cmdtext[2]==0){SendClientMessage(playerid,YOU_PICK_THE_COLOR,"Car chat: /c <text>"); return 1;}
  if(!IsPlayerInAnyVehicle(playerid)) {SendClientMessage(playerid,YOU_PICK_THE_COLOR,"You must be in a car to use car chat."); return 1;}

  new the_car = GetPlayerVehicleID(playerid);

  for(new player=0; player<MAX_PLAYERS; player++)
  {
     if(IsPlayerConnected(player) && IsPlayerInVehicle(player,the_car)) SendPlayerMessageToPlayer(player,playerid,cmdtext[3]);
   }
   return 1;
}
