Just a quick example.
pawn Code:
stock LoadServerVehicles()
{
    new
        idx = 1,
        query[128],
        Result[128];

    while(idx < MAX_VEHICLES)
    {
        format(query, sizeof(query), "SELECT * FROM `vehicles` WHERE `carid` = '%d'", idx );
        mysql_query(query);
        mysql_store_result();
        
        mysql_fetch_field("modelid", Result );
        modelid = strval( Result );

        mysql_free_result();
        idx++;
    }
}
