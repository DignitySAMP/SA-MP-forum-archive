public OnPlayerCommandText(playerid, cmdtext[])
{
    if(strcmp("/ice", cmdtext, true, 10) == 0)
	{
        if(GetVehicleModel(GetPlayerVehicleID(playerid)) == 423)
		{
            iceJob[playerid] = 1;
			new name[MAX_PLAYER_NAME], string[90];
			GetPlayerName(playerid, name, sizeof(name));
			format(string, sizeof(string), "* {6EF83C}%s {FFAF00}Sale {00C0FF}Ice Cream.", name );
			SendClientMessageToAll(COLOR_YELLOW, string);
			SetPlayerCheckpoint(playerid,1037.7463,-1333.4785,13.5476,3);
			SendClientMessage(playerid,COLOR_SILVER,"* Den checkpoint tren map!");
			return 1;
		}
        SendClientMessage(playerid, COLOR_RED,"Chi co o tren xe moi co the bat dau cong viec!");
    }
    return 0;
}
