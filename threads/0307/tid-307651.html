Script:

Code:
public CheckSQL()
{
	new bool:connected;
	if (!mysql_ping())
	{
		connected = false;
	}
	else
	{
		connected = true;
	}
	if (connected == false)
	{
		print("[MYSQL] Lost connection to MYSQL server, Reconnecting..");
		mysql_start();
	}
	return 1;
}
