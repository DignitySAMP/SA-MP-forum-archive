pawn Code:
#include <a_samp>
new barra;
#define VelocidadBarra (2.0)

public OnFilterScriptInit()
{
    barra = CreateObject(968, 1544.69, -1630.79, 13.07,   0.00, 90.00, 90.00);
    return 1;
}

public OnPlayerCommandText(playerid, cmdtext[])
{
    if (strcmp("/bajar", cmdtext, true, 10) == 0)
    {
        MoveObject(barra, 1544.69, -1630.79, 13.07-0.0001, VelocidadBarra, 0.00, 90.00, 90.00);
        return 1;
    }
    if (strcmp("/subir", cmdtext, true, 10) == 0)
    {
        MoveObject(barra, 1544.69, -1630.79, 13.07+0.0001, VelocidadBarra, 0.00, 0.00, 90.00);
        return 1;
    }
    return 0;
}
