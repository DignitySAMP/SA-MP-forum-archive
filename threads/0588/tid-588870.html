PHP Code:
#define PRESSED(%0) \
    (((newkeys & (%0)) == (%0)) && ((oldkeys & (%0)) != (%0)))
public OnPlayerKeyStateChange(playerid, newkeys, oldkeys)
{
if (PRESSED( KEY_YES ))
   {
    if(GetPlayerVirtualWorld(playerid) == 0) return CallRemoteFunction("OnPlayerCommandText","is",playerid,"/enter");
    else CallRemoteFunction("OnPlayerCommandText","is",playerid,"/exit");
   }
return 1;
}
