pawn Code:
public OnPlayerShootPlayer(Shooter,Target,Float:HealthLost,Float:ArmourLost)
{
    if(GetPVarInt(Shooter,"havepet") == 1)
    {
    HealthLost = HealthLost + 3;
    ArmourLost = ArmourLost + 2;
    }
    if(GetPVarInt(Target,"havepet") == 1)
    {
    HealthLost = HealthLost - 3;
    ArmourLost = ArmourLost - 2;
    }
    return 1;
}
