stock SalvarVip(playerid)// Stock de salvar VIP!
{
    if(PlayerVip[playerid] >= 1)// Se o cara for maior ou igual a 1
    {

            INI_Close();// Fecha o arquivo!
        }
    }
    return 1;
}
