2:Paste this command under onplayercommandtext
pawn Code:
if(!strcmp("/sniperdm", cmdtext, true))
    {
        new rand = random(sizeof(RandomDmSniperPos));
        SetPlayerPos(playerid, RandomDmSniperPos[rand][0], RandomDmSniperPos[rand][1],         RandomDmSniperPos[rand][2]);
        FreezeLoadObjects(playerid);
        new name[128];
        new string[256];
        GetPlayerName(playerid,name,128);
        format(string, sizeof(string), "%s go to /sniperdm by [ISS]jumbo",name);
        SendClientMessageToAll(0x99FFFFAA, string);
        SetPlayerInterior(playerid, 0);
        PlayerPlaySound(playerid, 1054, 0.0, 0.0, 0.0);
        return 1;
    }
now the function for load the map and not falling down
pawn Code:
forward FreezeLoadObjects(playerid);
public FreezeLoadObjects(playerid)
{
    TogglePlayerControllable(playerid,0);
    SetTimerEx("UnFreezeLoadObjects",3000,0,"i",playerid);
    GameTextForPlayer(playerid,"~r~Loading objects",2900,3);
}
forward UnFreezeLoadObjects(playerid);
public UnFreezeLoadObjects(playerid)
{
    TogglePlayerControllable(playerid,1);
    GameTextForPlayer(playerid,"~g~loaded",1000,3);
}
