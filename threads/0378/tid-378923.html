Изначально разделить параметры на 2 переменных, например.

new str1[64],str2[128];
sscanf(params,"s[64] s[128]",str1,str2);
if(!strlen(str1))return SendClientMessage(...);
if(!strcmp(str1,"repair"))
{
//--//
return 1;
}
if(!strcmp(str1,"lights"))
{
//--//
return 1;
}
if(!strcmp(str1,"fill"))
{
if(!strlen(str2))return SendClientMessage(...);
new litres=strval(str2);
//--//
return 1;
}
