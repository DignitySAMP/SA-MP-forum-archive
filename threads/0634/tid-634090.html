    for(new i=1; i < 11; i++)

        format(name, sizeof name, "name%d", i);
        format(points, sizeof points, "points%d", i);
        if(!strcmp(removename, DOF2_GetString(file, name), true)){
            temnick=true;
            DOF2_SetString(file,         name, "Ninguйm");
            DOF2_SetInt(file,             points, 0);

        }
    }
    if(temnick){
        DOF2_SaveFile();
    }
    return 1;
}
