Heres a breif example:
pawn Code:
public OnPlayerRequestSpawn(playerid)
{
  if(GetPlayerSkin(playerid) == /*skinid here*/)
  {
   if(strfind(name, "[MR]", true) == -1)
   {
     SendClientMessage(playerid, color, "Not allowed to use this skin!");
     return 0;
   }
  }
  return 1;
}
