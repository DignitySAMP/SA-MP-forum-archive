{

    {
        new string[368];

        strcat(string,"Hand Guns                                    [$1,000]\n");
        strcat(string,"Shotguns                                       [$5,000]\n");
        strcat(string,"Assault Weapons                          [$8,000]\n");

        ShowPlayerDialog(playerid, DIALOG_WEAPONS, DIALOG_STYLE_LIST, "Weapons:",string,"Select","Back");

    return 0;
}
