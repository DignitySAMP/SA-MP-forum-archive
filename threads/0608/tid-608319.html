Simples, utilize PVars!

PHP Code:
public OnDialogResponse(playerid, dialogid, response, listitem, inputtext[])
{
    if(dialogid == 7000)
    {
        
        // Criamos uma PVar (player variable:https://wiki.sa-mp.com/wiki/SetPVarInt)
        SetPVarInt(playerid, "lojaBALAS", strval(inputtext0));
        
        ShowPlayerDialog(playerid, 7001, DIALOG_STYLE_INPUT, "Quantidade de Balas", "Digite abaixo a quantidade de balas:", "Comprar", #);
    }
    if(dialogid == 7001)
    {
        if(response)
        {
            // Agora iremos pegar a nossa PVar que criamos anteriormente com a funзгo GetPVarInt(https://wiki.sa-mp.com/wiki/GetPVarInt)
            GivePlayerWeapon(playerid, GetPVarInt(playerid, "lojaBALAS"), strval(inputtext));
            
            // Deletando nossa PVar (https://wiki.sa-mp.com/wiki/DeletePVar)
            DeletePVar(playerid, "lojaBALAS");
        }
    }
    return 1;
}
