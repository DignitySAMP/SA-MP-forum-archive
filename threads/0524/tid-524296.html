public OnPlayerCommandText(playerid, cmdtext[])
{
    if (strcmp("/mp3", cmdtext, true, 10) == 0)
    {
        new dialog[1200];
        strcat(dialog,"Saturday Saturday\nCaller Tune\nPiya Kay Bazaar\nJust Look Into My Eyes\nBarbaad Raat\n");
        strcat(dialog,"Hum Pagal Nahi Hain\nKholday Dil Ki Khidki\nWhistle Bja\nRabba\nRaat Bhar\nTabah\n");
        strcat(dialog,"Pappi Song\nRabah Main\nTonight is the Night\nHooligans Remix\nGalliyan\nFireWorks\nGive Me EveryThing Tonight\n");
        strcat(dialog,"Just The Way You are\nMarry You\nWhats My Name\nThats My Name\nEarth Quake\nPass Out\nWritten In Stars\nRolling in the deep\n");
        strcat(dialog, "Some One like you\nStereo Hearts\nWild Ones\nZaroorat\nMain Dhundne ko zamaane mai\nStop music");
        SendClientMessage(playerid,COLOR_LIGHTGREEN,"Welcome To Music Station New Vegas Cops And Robbers");
        ShowPlayerDialog(playerid, 1, DIALOG_STYLE_LIST, "{FFFF00}Music Station",dialog,"Ok","Exit");
        return 1;
    }
    return 0;
}
