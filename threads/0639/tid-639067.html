{

    {X,Y,Z},//pizza local 9
    {X,Y,Z}//pizza local 10
};
CMD:pegarpizza(playerid)
{
    if(CountPizzas[playerid]==10)return SendClientMessage(playerid, -1,"vc jб pegou o mбximo de pizzas. Tente mais tarde");
    if(!IsPlayerInRangeOfPoint(playerid, 7.0, PIZZA_LOCAL))return SendClientMessage(playerid, -1,"Vc nгo estб numa pizzaria");
    EntregandoPizzas=1;
    SetPlayerCheckpoint(playerid, ChecksPizzas[rand][0], ChecksPizzas[rand][1],ChecksPizzas[rand][2], 5.0);
    return 1;
}
public OnPlayerEnterCheckpoint(playerid)
{
    if(EntregandoPizzas[playerid]==1)
    {
        CountPizzas[playerid]++;
        DisablePlayerRaceCheckpoint(playerid);

    }
