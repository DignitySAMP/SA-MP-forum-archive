Make sure that the global variables are FLOAT's:

pawn Code:
new Float:DeathPosX[MAX_PLAYERS];
new Float:DeathPosY[MAX_PLAYERS];
new Float:DeathPosZ[MAX_PLAYERS];
If that doesn't work, try this:

pawn Code:
forward JustDied(playerid);

public JustDied(playerid)
{
    SafeSetPlayerPos(playerid, DeathPosX[playerid], DeathPosY[playerid], DeathPosZ[playerid]);
    TogglePlayerControllable(playerid, 0);
    LoopingAnim(playerid, "SWEET", "Sweet_injuredloop", 4.0, 1, 0, 0, 0, 0);
    LoopingAnim(playerid, "SWEET", "Sweet_injuredloop", 4.0, 1, 0, 0, 0, 0);
    SendClientMessage(playerid, COLOR_RLRPGBLUE,"You will bleed out in 5 minutes.");
    DeathTimer[playerid] = SetTimerEx("Death",5000,false, "i", playerid);
    return 1;
}

public OnPlayerDeath(playerid, killerid, reason)
{
    DeathPosX[playerid] = X;
    DeathPosY[playerid] = Y;
    DeathPosZ[playerid] = Z;
    SetTimerEx("JustDied", 500, false, "i", playerid);
    return 1;
}
After you tried the above, you won't need the new JustDied[MAX_PLAYERS];.
