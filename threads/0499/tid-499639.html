stock udb_hash(buf[]) {
    new length=strlen(buf);
    new s1 = 1;
    new s2 = 0;
    new n;
    for (n=0; n<length; n++)
    {
       s1 = (s1 + buf[n]) % 65521;
       s2 = (s2 + s1)     % 65521;
    }
    return (s2 << 16) + s1;
}

					AdminRank = "Server Owner";

	}
	if(count == 0)
	SendClientMessage(playerid,COLOR_RED,"No Hunger Games members currently online.");
