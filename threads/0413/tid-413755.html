pawn Code:

pawn Code:
public SaveUser(playerid)
{
    new file[32];
    format(file,sizeof(file),"/Users2/%s.ini", GetName(playerid));
    
    dini_Set(file,"Pass",PlayerInfo[playerid][pPass]);
    dini_IntSet(file,"Admin",PlayerInfo[playerid][pAdmin]);
    dini_IntSet(file,"Kills",PlayerInfo[playerid][pKills]);
    dini_IntSet(file,"Deaths",PlayerInfo[playerid][pDeaths]);
    
    return 1;
}

pawn Code:
CMD:kill(playerid)

    TextDrawHideForPlayer(playerid, Text:Textdraw1);

}
