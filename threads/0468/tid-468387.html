#include <a_samp> //principal
#include <icmd> //processador de comando.

#define DIALOG_RADIO 2212  //id do nosso dialog

public OnFilterScriptInit() return print("FS-Radio ====Carregado====!");
public OnFilterScriptExit() return print("FS-Radio ====Descarregado====!");

public OnDialogResponse(playerid, dialogid, response, listitem, inputtext[])
{
    switch(dialogid)
    {
        case DIALOG_RADIO:
        {
            if(response)
            {
                if(listitem == 0)
                    PlayAudioStreamForPlayer(playerid, "http://188.241.154.61:6060/"); //caso ele escolha a primeira opзгo
                if(listitem == 1)
                    PlayAudioStreamForPlayer(playerid, "http://94.23.51.96:8000/"); //caso ele escolha 2 opзгo
                if(listitem == 2)
                    PlayAudioStreamForPlayer(playerid, "http://91.121.174.35:8000/");//caso ele escolha 3 opзгo
                if(listitem == 3)
                    PlayAudioStreamForPlayer(playerid, "http://198.50.197.162:8024/");//caso ele escolha 4 opзгo
                if(listitem == 4)
                    PlayAudioStreamForPlayer(playerid, "http://194.232.200.156:8000/");//caso ele escolha 5 opзгo
                if(listitem == 5)
                    PlayAudioStreamForPlayer(playerid, "http://198.178.123.20:7000/");//caso ele escolha 6 opзгo
                if(listitem == 6)
                    PlayAudioStreamForPlayer(playerid, "http://108.61.73.115:8054/");//caso ele escolha 7 opзгo
                if(listitem == 7)
                    PlayAudioStreamForPlayer(playerid, "http://192.81.248.32:8557/");//caso ele escolha 8 opзгo
                if(listitem == 8)
                    PlayAudioStreamForPlayer(playerid, "http://222.122.131.58:6400/");//caso ele escolha 9 opзгo
                if(listitem == 9)
                    PlayAudioStreamForPlayer(playerid, "http://174.36.42.110:18110/");//caso ele escolha a ultima opзгo
                if(listitem == 10)
                    StopAudioStreamForPlayer(playerid);
            }
            return 1;
        }
    }
    return 1;
}

public OnPlayerCommandText(playerid, commandid, params[]) {
    switch(commandid) {
        case iscmd("r,a,d,i,o,s"): {
            if(!params[0]) {
                ShowPlayerDialog(playerid, DIALOG_RADIO, DIALOG_STYLE_LIST, "Radios", "Radio HipHop\nRadio Classic\nRadio Rock\nRadio Eletronica\n Radio Pop\n\
                Radio Raggae\nRadio Rap\nRadio Punk\nRadio in Piano\nRadio Hard Rock\nDesligar Radio", "Selecionar", "Sair");

            }
            return 1;
        }
        default: return SendClientMessage(playerid, -1, "[Erro]: Comando desconhecido, desculpe."); //se ele digita comando errado
    }
    return 0;
}
