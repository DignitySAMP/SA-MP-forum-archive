while (paramPos < paramCount && string[stringPos])

case 'i', 'd':
{
new
neg = 1,

ch = string[stringPos];

{
neg = -1;
ch = string[++stringPos];

do
{
stringPos++;
if (ch >= '0' && ch <= '9')
{
num = (num * 10) + (ch - '0');
}
else
{
return 1;
}
}
while ((ch = string[stringPos]) && ch != ' ');
setarg(paramPos, 0, num * neg);
}
case 'h', 'x':
{
new
ch,

{
switch (ch)
{
case 'x', 'X':

num = 0;
continue;
}
case '0' .. '9':
{
num = (num << 4) | (ch - '0');
}
case 'a' .. 'f':
{
num = (num << 4) | (ch - ('a' - 10));
}
case 'A' .. 'F':
{
num = (num << 4) | (ch - ('A' - 10));

case ' ':

default:

}
setarg(paramPos, 0, num);

strmid(tmp, string, stringPos, stringPos+sizeof(tmp)-2);

case 's', 'z':
