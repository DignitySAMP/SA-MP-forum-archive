stock PlayerCar_Unload(playerid)
{
    for(new id; id < MAX_DYNAMIC_CARS; id++)
    {
        if (IsVehicleSpawned(id))
        {
            if (PlayerData[playerid][pID] == CarData[id][carOwner])
            {
                PlayerCar_Save(id);
                DestroyVehicle(id);
                CarData[id][carExists] = false;
            }
        }
    }
    return 1;
}
