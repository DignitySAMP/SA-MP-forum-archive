How To Make Textdraw Please Help Me
Code:
new Text:Value[MAX_PLAYERS];

public OnGameModeInit()
{
	for(new i=NULL; i<MAX_PLAYERS;i++)  // at the end of the function move it  OnGameModeInit()
	{
		void[i] = TextDrawCreate(270.0, 200.0, "");
		TextDrawFont(Value[i], 1);
		TextDrawColor(Value[i], 0x00FF00FF);
		TextDrawSetOutline(Value[i],0);
		TextDrawLetterSize(Value[i], 0.250000, 1.200000);
		TextDrawSetProportional(Value[i], 1);
		TextDrawSetShadow(Value[i], 1);
	}

return true;
}


public OnPlayerDeath(playerid, killerid, reason)
{
	if(killerid != INVALID_PLAYER_ID)
	{
	TextDrawSetString(Value[killerid],"+$2.500~n~~y~~l~+1 Score");  //here you can add different amounts of money to different weapons, use format.
	TextDrawShowForPlayer(playerid, Value[killerid]);
        SetTimerEx("KillTextDraw", 1000, false, "i", killerid);
	}
	return true;
}

forward KillTextDraw(killerid);
public KillTextDraw(killerid) return TextDrawHideForPlayer(killerid, Value[killerid]);


public OnPlayerDisconnect(playerid, reason)
{

	TextDrawHideForPlayer(playerid, Value[playerid]);
        return true;
}

PHP Code:
new PlayerText:Value[MAX_PLAYERS];
    
public OnPlayerConnect(playerid)
{
    Value[playerid] = TextDrawCreate(270.0, 200.0, "");
    PlayerTextDrawFont(Value[playerid], 1);
    PlayerTextDrawColor(Value[playerid], 0x00FF00FF);
    PlayerTextDrawSetOutline(Value[playerid],0);
    PlayerTextDrawLetterSize(Value[playerid], 0.250000, 1.200000);
    PlayerTextDrawSetProportional(Value[playerid], 1);
    PlayerTextDrawSetShadow(Value[playerid], 1);
}
public OnPlayerDeath(playerid, killerid, reason)
{
    if(killerid != INVALID_PLAYER_ID)
    {
        PlayerTextDrawSetString(Value[killerid],"+$2.500~n~~y~~l~+1 Score");  //here you can add different amounts of money to different weapons, use format.
        PlayerTextDrawShow(playerid, Value[killerid]);
        GivePlayerMoney(playerid,2500);
        SetTimerEx("KillTextDraw", 1000, false, "i", killerid);
    }
    return true;
}
forward KillTextDraw(killerid);
public KillTextDraw(killerid) return PlayerTextDrawHide(killerid, Value[killerid]);
public OnPlayerDisconnect(playerid, reason)
{
        if(Value[playerid] != Text:INVALID_TEXT_DRAW) PlayerTextDrawDestroy(playerid, Value[playerid]);
        return true;
}
