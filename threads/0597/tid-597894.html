Code:
CMD:stats(playerid, params[])
{
	new giveid;
	if(!sscanf(params, "u", giveid))
	{
		if(IsPlayerConnected(giveid))
		{
			ShowPlayerDialog(playerid, 4, DIALOG_STYLE_MSGBOX, "{D1D1D1}Player Stats", ShowStats(giveid), "Ok", "");

}
