pawn Code:
new
    bool: Saved_Pos[ MAX_PLAYERS char ],
    Float: Saved_X[ MAX_PLAYERS ],
    Float: Saved_Y[ MAX_PLAYERS ],
    Float: Saved_Z[ MAX_PLAYERS ],
    Float: saved_Angle[ MAX_PLAYERS ]
;

public OnPlayerSpawn( playerid )
{
    if( Saved_Pos{ playerid } )
    {
        SetPlayerPos( playerid, Saved_X[ playerid ], Saved_Y[ playerid ], Saved_Z[ playerid ] );
        SetPlayerFacingAngle( playerid, Saved_Angle[ playerid ] );
    }
    else
    {
        // set the position for those who has not saved
    }
    return 1;
}

// Command /spawn
GetPlayerPos( playerid, Saved_X[ playerid ], Saved_Y[ playerid ], Saved_Z[ playerid ] );
GetPlayerFacingAngle( playerid, Saved_Angle[ playerid ] );
Saved_Pos{ playerid } = true;

// NOTE1: When they disconnect, save the position (X, Y, Z, Angle) to the file as well as whether they've a saved position
// NOTE2: When they loggin, assign the floats to the variables above and also to the Saved_Pos
// NOTE3: Do not forget to reset the variables on connect

There are many tutorials about how to use y_ini. Read some of them!

By the way, the coordinates are floats thus it should be:

INI_WriteFloat
INI_Float

The above was for integers and actually are not useful to your case. The boolean can be used with INI_Bool/INI_WriteBool.

For more information, go: https://forum.sa-mp.com/showthread.php?tid=175565
