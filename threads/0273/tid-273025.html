stock token_by_delim(const string[], return_str[], delim, start_index)
{
    new x=0;
    while(string[start_index] != EOS && string[start_index] != delim) {
        return_str[x] = string[start_index];
        x++;
        start_index++;
    }
    return_str[x] = EOS;
    if(string[start_index] == EOS) start_index = (-1);
    return start_index;
}

pawn Code:
stock LoadClasses(Path[])
{
    new line[64], File:Fhnd, Float:P[4], Ext[3];
    if(!fexist(Path))return 0;
    Fhnd = fopen(Path, io_read);
    while(fread(File, line){
        sscanf(line, "dffffdd", Ext[0], P[0], P[1], P[2], P[3], Ext[1], Ext[2]);
        AddPlayerClass(Ext[0], P[0], P[1], P[2], P[3], Ext[1], Ext[2]);
    }
    return 1;
}
El formato en el que tenes que guardar los datos para que ese codigo funcione es el siguiente:
Skin Coordenada:X Coordenada:Y Coordenada:Z Angulo Arma1 MunicionesDelArma1
(sin comas, solo los valores uno al lado del otro separados por un espacio)

PD: para los codigos usa [pawn*] & [/pawn*] (sin los '*')
