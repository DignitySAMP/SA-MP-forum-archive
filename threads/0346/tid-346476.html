stock split(const strsrc[], strdest[][], delimiter){
    new pi, li, aNum, len;
    while(pi<=strlen(strsrc)){
        if(strsrc[pi]==delimiter || pi==strlen(strsrc)){
            len=strmid(strdest[aNum], strsrc, li, pi, 128);
            strdest[aNum][len]=0;
            li=pi+1;
            aNum++;
        }
        pi++;
    }
