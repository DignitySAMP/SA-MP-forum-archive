pawn Code:
// Putting this at top
#define Clear(%0)   for(new xz = 0; xz < 100; xz++) SendClientMessage(%0, -1, " ")

if (strcmp("/flush", cmdtext, true, 5) == 0)
{
    Clear(playerid);
    return true;
}

// ZCMD is recommended - strcmp is slow.
CMD:flush(playerid)
{
      Clear(playerid);
      return true;
}
