public OnPlayerDisconnect(playerid, reason)
{
    SaveGunStats(playerid);
    return 1;
}

public OnPlayerConnect(playerid)
{
    if(fexist(GunPath(playerid)))
    {
        INI_ParseFile(GunPath(playerid), "LoadGun_%s", .bExtra = true, .extra = playerid);
    }
    return 1;
}
