    gTeamSelection[ playerid ] = true;

	TextDrawHideForPlayer( playerid, TextAfk2 );
	TextDrawHideForPlayer( playerid, Name_Intro );
	TextDrawHideForPlayer( playerid, Reaction_TD );
	
	PlayerTextDrawHide( playerid, gTerswarTD[ playerid ] );
