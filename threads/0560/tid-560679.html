forward Hash(string[]);
public Hash(string[])
{
	for(new x=0; x < strlen(string); x++)
	{
	    string[x] += (3^x) * (x % 15);
		if(string[x] > (0xff))
		{
		    string[x] -= 256;
	    }
    }
	return 1;
}
