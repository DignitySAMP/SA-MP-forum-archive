PHP Code:
public OnGameModeInit()
{
    Position = TextDrawCreate(1.0, 5.6, "Welcome to the point"); 
    return 1;
}
if(IsPlayerInRangeOfPoint(playerid, 7.0, 93.6235,1921.1 780,18.0469))
    {
        TextDrawShowForPlayer(playerid,Position);
    }
