pawn Code:
new TimesCalled = 0;
Put this on ongamemodeinit
pawn Code:
SetTimer("GiveNewWeapons", 60000*10 /*10 minutes*/ , true)
this is the callback that happens every 10 minutes [you can do anything here like set all players to respawn or something]
pawn Code:
public GiveNewWeapons()
{
  for(new i = 0; i < MAX_PLAYERS; i++)
  {
      if(TimesCalled == 0)
      {
        GivePlayerWeapon(i, startingweapon, ammo);
        ResetPlayerWeapons(i);
      }
      if(TimesCalled == 1)
      {
        GivePlayerWeapon(i, newweapon, ammo);
        ResetPlayerWeapons(i);
      }
      if(TimesCalled == 2)
      {
        GivePlayerWeapon(i, newweapon, ammo);
        ResetPlayerWeapons(i);
      }
      if(TimesCalled == 3)
      {
        GivePlayerWeapon(i, newweapon, ammo);
        ResetPlayerWeapons(i);
      }
      if(TimesCalled == 4)
      {
        GivePlayerWeapon(i, newweapon, ammo);
        ResetPlayerWeapons(i);
      }
      // add more if you want . . .
/*
      if(TimesCalled == anothernumber)
      {
        GivePlayerWeapon(i, anotherweapon, ammo);
        ResetPlayerWeapons(i);
      }
*/
      if(TimesCalled == 5)
      {
        TimesCalled = 0; // reset the counter so it goes back to first weapons and maybe restart the round or something
        //RestartRound(whatever...);
        ResetPlayerWeapons(i);
      }
  }
    TimesCalled++;
}
