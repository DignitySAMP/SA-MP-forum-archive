forward SaveToFile(filename[],text[]);
public SaveToFile(filename[],text[])
{
    new File:FileS, filepath[128], string[128], year,month,day, hour,minute,second;
    getdate(year,month,day); gettime(hour,minute,second);

    format(filepath,sizeof(filepath),"logs/%s.txt",filename);
    FileS = fopen(filepath,io_append);
    format(string,sizeof(string),"[%d.%d.%d %d:%d:%d] %s\r\n",day,month,year,hour,minute,second,text);
    fwrite(FileS,string);
    fclose(FileS);
    return 1;
}

forward MessageToAdmins(color,const string[]);
public MessageToAdmins(color,const string[])
{

    {
        if(IsPlayerConnected(i) == 1) if (IsPlayerAdmin(i)) SendClientMessage(i, color, string);
    }
    return 1;
}
