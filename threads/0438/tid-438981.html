{
    dead[playerid] = 0;

    GetPlayerName(playerid,name,sizeof(name)); //Get player's name

    {// then
        INI_ParseFile(Path(playerid),"loadaccount_%s", .bExtra = true, .extra = playerid); //Will load user's data using INI_Parsefile.

        return 1;
    }
    

    
    return 1;
}
