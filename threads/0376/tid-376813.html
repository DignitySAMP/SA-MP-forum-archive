public OnPlayerCommandText(playerid, cmdtext[])
{
    if (strcmp(cmdtext,"/pararmusica",true) == 0)
    {
        SendClientMessage(playerid, 0x0000FFFF, "Vocк parou a mъsica digite /MP3 novamente para selecionar outra.");
        StopAudioStreamForPlayer(playerid);

    if (strcmp(cmdtext,"/mp3",true) == 0)
    {
        new musica[1000];

    return 0;
}
