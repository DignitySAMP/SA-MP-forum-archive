                    GivePlayerWeapon(playerid,x,9999);
                    GivePlayerWeapon(playerid,x,9999);
                    GivePlayerWeapon(playerid,x,9999);

            }
                case 1: // Triads
            {
                    SetPlayerSkin(playerid,x);
                    GivePlayerWeapon(playerid,x,9999);
                    GivePlayerWeapon(playerid,x,9999);
                    GivePlayerWeapon(playerid,x,9999);

            }
                case 2: // Russian Mafia
            {
                   SetPlayerSkin(playerid,x);
                   GivePlayerWeapon(playerid,x,9999);
                   GivePlayerWeapon(playerid,x,9999);
                   GivePlayerWeapon(playerid,x,9999);
                   SetPlayerPos(playerid,x,y,z);
            }
                case 3://Italian Mafia
            {
                   SetPlayerPos(playerid,x,y,z);
                   SetPlayerSkin(playerid,x);
                   GivePlayerWeapon(playerid,x,9999);
                   GivePlayerWeapon(playerid,x,9999);
                   GivePlayerWeapon(playerid,x,9999);
            }
        }
        return 1;
    }
}
