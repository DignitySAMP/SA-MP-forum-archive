pawn Code:
new plTimer[MAX_PLAYERS], plSeconds[MAX_PLAYERS];

public OnPlayerDisconnect(playerid, reason){
    KillTimer(plTimer[playerid]);
    return 1;
}

public OnPlayerConnect(playerid){
    plTimer[playerid] = SetTimerEx("secondTimer", 1000, true, "i", playerid);
    plSeconds[playerid] = 0;
    return 1;
}

forward secondTimer(playerid);
public secondTimer(playerid){
    new str[30];
    plSeconds[playerid] ++;
    
    format(str, sizeof(str), "%d seconds", plSeconds[playerid]);
    GameTextForPlayer(playerid, str, 3000, 5);
    return 1;
}
There's an example for you.
