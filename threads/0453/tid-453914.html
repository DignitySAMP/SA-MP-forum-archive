pawn Code:
// on player connect
   TextDrawShowForPlayer(playerid, Black1), TextDrawShowForPlayer(playerid, Black2);

// on player spawn 
   TextDrawHideForPlayer(playerid, Black1), TextDrawHideForPlayer(playerid, Black2);
