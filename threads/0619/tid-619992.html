Implementation:
PHP Code:
// Loading:
for(new i, j = cache_get_row_count(); i < j && i < MAX_VEHICLES; i++) {
    if(!VehicleExists[i]) {
        VehicleExists[i] = true;
        // Rest of data assignments
    }
}
// Creation:
for(new i; i < MAX_VEHICLES; i++) if(!VehicleExists[i]) {
    // Rest of data assignments
}
