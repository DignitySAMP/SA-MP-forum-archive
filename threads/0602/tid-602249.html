stock SaveFactions()
{
	// LSPD
	new file[128];
	format(file, sizeof(file), "factions/LSPD.ini");
	if(!dini_Exists(file)) dini_Create(file);
	dini_Set(file, "LSPDLeader", FacInfo[1][facLeader]);
	dini_Set(file, "LSPDMembers", FacInfo[1][facMembers]);
