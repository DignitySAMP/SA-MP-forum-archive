pawn Code:
stock Float: floatrandom(Float: fMin, Float: fMax, iDig)
{
    iDig = _: floatpower(10.0, iDig);
    fMin = Float: (floatround(fMin * Float: iDig));
    fMax = Float: (floatround(fMax * Float: iDig));
    return (random((_: fMax - _: fMin)) + _: fMin) / Float: iDig;
}
