If you have a question : How to get inside the interior ?.
I will answer : Well, its easy you only need to put this into your Gamemode.

Code:
// Put this on OnGamemodeInIt
CreatePickup(1239,1,2496.2759,-26.8295,28.0043);
CreatePickup(1239,1,2502.3994,-26.8187,28.0043);
Code:
public OnPlayerPickUpPickup(playerid, pickupid)
{
    GameTextForPlayer(playerid,"~r~ /enter ~y~ to enter this building",3000,1);
    return 1;
}
Code:
public OnPlayerCommandText(playerid, cmdtext[])
{	
    if (strcmp("/enter", cmdtext, true, 6) == 0)
    {
        if(IsPlayerInRangeOfPoint(playerid,10,2496.2759,-26.8295,28.0043))
            {
                SetPlayerPos(playerid,2499.2771,-31.1802,27.9850);
                return 1;
            }
    }
    if (strcmp("/enter", cmdtext, true, 6) == 0)
    {
        if(IsPlayerInRangeOfPoint(playerid,10,2502.3994,-26.8187,28.0043))
            {
                SetPlayerPos(playerid,2499.2771,-31.1802,27.9850);
                return 1;
            }
    }
    return 0;
}
