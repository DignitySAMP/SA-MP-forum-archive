public IsPlayerAtToll()
{
    for(new playerid=0; playerid<MAX_PLAYERS; playerid++)
    {
        if(IsPlayerInRangeOfPoint(playerid,7.0,1623.5999755859,-2163.8999023438,14.199999809265))
        {
            MoveObject(garagem,1614.9000244141,-2163.6000976563,14.199999809265,3.0);
            SetTimerEx("Closegaragem", 10000, false, "i", garagem);
        }
    }
    return 1;
