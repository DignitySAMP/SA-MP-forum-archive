public OnGameModeExit()
{
    for(new i=0;i<MAX_PLAYERS;i++)
        if(IsPlayerConnected(i))
            StopPlayerHoldingObject(i);

    if(isLogged[i] == 1) //SAVE ON CRASH/RESTART
    {
    new name[256],check[256];
    Delete3DTextLabel(levl[i]);
