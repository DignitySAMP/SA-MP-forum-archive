//============================================================================ > udb_hash

stock udb_hash(buf[]) {
    new length=strlen(buf), s1 = 1, s2 = 0, n;
    for (n=0; n<length; n++) { s1 = (s1 + buf[n]) % 65521; s2 = (s2 + s1)     % 65521; }
    return (s2 << 16) + s1; }
