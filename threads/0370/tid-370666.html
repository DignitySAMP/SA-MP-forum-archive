{

    new vehicleid,Float:speed_x,Float:speed_y,Float:speed_z,Float:final_speed,speed_string[256],final_speed_int;

            GetVehicleVelocity(vehicleid,speed_x,speed_y,speed_z);

        else {
        TextDrawSetString(VStats, " ");
