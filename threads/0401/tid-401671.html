stock IsBanned(ip[])
{
    new BigString[1234];
    new File:File = fopen("Bans.txt",io_read);
    while(fread(File,string))
    {
        if(strfind(string,ip))
        {
            printf("%s is banned.",ip);
        }
        else
        {
            printf("%s is not banned.",ip);
    }
    return 1;
}
