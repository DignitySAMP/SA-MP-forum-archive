{

			//------------------------------------------------------------------
		}
		else
		{
			format(eString, sizeof(eString), "{FF0000}Autentificare nereusita (%d/%d)!\n\n{FFFF00}Ai introdus parola gresita. Te rugam sa incerci din nou!", PlayerInfo[playerid][pFailLogin], MaxWarnings);
