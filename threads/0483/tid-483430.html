A Command to go inside the Bar
Code:
 public OnPlayerCommandText(playerid, cmdtext[])
{
 if(strcmp("/gotobar", cmdtext, true) == 0)
    {
        SetPlayerPos(playerid, 1472.2490, -1371.6003, 62.4375);
        SetPlayerFacingAngle(playerid, 180);
        SendClientMessage(playerid, 0x00FFFFAA, "You have been Teleported to Los Santos Bar.");
 }
 return 1;
}
