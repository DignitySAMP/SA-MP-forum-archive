1. Indentation is your friend, use it
2. Missing brackets, missing brackets everywhere - you need to learn the basics
3. Use [ pawn] tag
4. Are you writing a gamemode or filterscript? I assume gamemode for now
5. Writing commands using strcmp is currently really discouraged. Use ZCMD (https://forum.sa-mp.com/showthread.php?tid=91354) or command system from awegoddamnitsome ysi package ( https://forum.sa-mp.com/showthread.php?tid=169029 / https://forum.sa-mp.com/showthread.php?tid=321092 )

public OnPlayerEnterCheckpoint(playerid)
{
    DisablePlayerCheckpoint(playerid);
    if(GetVehicleModel(GetPlayerVehicleID(playerid)) == 455) {

            SendClientMessage(playerid, COLOR_RED, "Loading Rubbish");
            new rand = random(sizeof(checkpoints));
            SetPlayerCheckpoint(playerid, checkpoints[rand][0], checkpoints[rand][1], checkpoints[rand][2], checkpoints[rand][3]);

            MissionStage[playerid] = STAGE_ONROUTE;
        }
        if(STAGE_ONROUTE == MissionStage[playerid]) {
            SendClientMessage(playerid, COLOR_RED, "Unloading Rubbish");
            GameTextForPlayer(playerid, "~g~Yo Completed The Mission", 3000, 3);
            SetPlayerScore(playerid, GetPlayerScore(playerid) + 1);
            GivePlayerMoney(playerid, 5000);
            MissionStage[playerid] = STAGE_NONE;
        }
    }
    return 1;
