			    DarDinero(playerid, 5000);
			    QuitarArmas(playerid);

			    SendClientMessageToAll(-1, v_var);

			    KillTimer(InWater_BR[playerid]);
				KillTimer(Partida_BR);
				Timer_BT = SetTimer("IniciarBR", 30000, true);

				Hour = 0;
				Min = 0;
				E_BR_PLAYERS = 0;
				E_BR_CLOSE = 0;
				BR_Atrasado[playerid] = false;

	    }
	}
	return 1;
}
