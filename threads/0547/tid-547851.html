Boa tarde,queria ver se alguem pudesse me dizer como eu faria para que quando eu fosse para o mundo mata mata o id 5 e depois voltasse para qualquer outro mundo e as armas se deletassem sendo assim so apareзeriam no mundo mata mata,como faria codigo abaixo!!

pawn Code:
CMD:mundo(playerid, params[])
{
    new mund;
    if(sscanf(params, "d",mund))
        return SendClientMessage(playerid, -1, "[USO] /mundo[0-1000]");

    if(mund < 0 || mund > 1000)
        return SendClientMessage(playerid, -1, "[USO] /mundo[0-1000] *minino 0 e mбximo 1000*");

    if(IsPlayerInAnyVehicle(playerid)){
        new car = GetVehicleModel(GetPlayerVehicleID(playerid));
        if(car == 520 || car == 432 || car == 592 || car == 425 || car == 464 || car == 476 || car == 447) DestroyVehicle(GetPlayerVehicleID(playerid));
    }
    if(mund == 4) { //gravaзхes
        Msg(playerid, -1, "vocк foi para o mundo gravaзхes!!");
        SetPlayerVirtualWorld(playerid, mund);
    }
    if(mund == 5) { // mata mata
        GivePlayerWeapon(playerid,24,10000);
        GivePlayerWeapon(playerid,25,10000);
        GivePlayerWeapon(playerid,32,10000);
        SetPlayerHealth(playerid, 100);
        SetPlayerArmour(playerid, 100);
        Msg(playerid, -1, "vocк foi para o mundo mata mata!!");
        SetPlayerVirtualWorld(playerid, mund);
    }

    if(IsPlayerInAnyVehicle(playerid) && GetPlayerState(playerid) == 2) {
        SetVehicleVirtualWorld(GetPlayerVehicleID(playerid), mund);
    }
    format(String, 50, "Vocк foi para o mundo %d", mund);
    Msg(playerid, -1, String);
    SetPlayerVirtualWorld(playerid, mund);
    SetPVarInt(playerid, "mundo", mund);
    Mundo[playerid] = mund;
    new pname[MAX_PLAYER_NAME];
    GetPlayerName(playerid, pname, MAX_PLAYER_NAME);
    format(String, sizeof(String), ":: %s Foi para outro mundo virtual (/Mundo)", pname);
    MsgToAll(Verde, String);
    return true;
}
