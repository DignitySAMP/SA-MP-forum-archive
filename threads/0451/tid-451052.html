public OstecenjeVozila()
{
	for(new i = 0; i < MAX_VEHICLES; i++)
	{
		if(GetPlayerState(i) == PLAYER_STATE_DRIVER)
		{
			new Float:HP;
			GetVehicleHealth(GetPlayerVehicleID(i), HP);
			if(HP < 400.0)
			{
				new engine, lights, alarm, doors, bonnet, boot, objective;
				GetVehicleParamsEx(GetPlayerVehicleID(i), engine, lights, alarm, doors, bonnet, boot, objective);
				SetVehicleParamsEx(GetPlayerVehicleID(i), 0, lights, alarm, doors, bonnet, boot, objective);
				TextDrawShowForPlayer(i, TDSpeedo4);
				TextDrawHideForPlayer(i, TDSpeedo3);
			}
		}
		if(!IsPlayerInAnyVehicle(i)) TextDrawHideForPlayer(i, TDSpeedo1[i]), TextDrawHideForPlayer(i, TDSpeedo2[i]), TextDrawHideForPlayer(i, TDSpeedo3), TextDrawHideForPlayer(i, TDSpeedo4);
	}
	return 1;
}

It is this:

for(new i = 0; i < MAX_VEHICLES; i++)


Should be this:

for(new i = 0; i < MAX_PLAYERS; i++)

You got the errors for this reason, MAX_PLAYERS is defined as 500, and MAX_VEHICLES as 2000, so it was trying to go above 500 in the loop which was causing debug errors
