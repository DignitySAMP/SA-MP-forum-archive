At top of your script
pawn Code:
new messagenumber = 0;
new messages[][] =
    {
        "~Msg1",
        "~msg2",
        "msg3"
    };
OnGameModeInIt
pawn Code:
SetTimer("SendMessages",60000, true);
Anywhere "NOT IN CALLBACKS"
pawn Code:
forward SendMessages();
public SendMessages()
{
    if(messagenumber == sizeof(messages)) messagenumber = 0;
    SendClientMessageToAll(-1, messages[messagenumber]);
    messagenumber ++;
    return 1;
}
