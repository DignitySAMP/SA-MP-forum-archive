*Skydive challenge!*
This is a rather hard challenge i've made. It's old but my decision is to release it now.
Many players have told it's almost impossible so you might want to change the reward to something better.
Before you jump of the edge make sure to zoom out your camera, and then move slowly while diving down, otherwise you lose control and die.

*Notes*
A streamer is recommended, i use Incognito's streamer.

*Video*
[ame]http://www.youtube.com/watch?v=4XDyP92dbag&hd=1[/ame]

Enjoy.

public OnPlayerPickUpPickup(playerid, pickupid)
{
	if(pickupid == divesouth)
	{
		GameTextForPlayer(playerid,"~r~~h~Congrats!~n~~w~You finished the challenge.",6000,0);
		PlayerPlaySound(playerid,1149,0,0,0);
		GivePlayerWeapon(playerid,24,5000);
		GivePlayerWeapon(playerid,26,5000);
		GivePlayerWeapon(playerid,28,5000);
		GivePlayerWeapon(playerid,35,10);
		GivePlayerWeapon(playerid,18,10);
		SendClientMessage(playerid,0x7CFC00AA,"Reward: Weapons, RPG and Molotovs.");
    	new string[64];
    	new name[32];
    	GetPlayerName(playerid,name,128);
    	format(string, sizeof(string), "%s finished the challenge: Dive!",name);
		SendClientMessageToAll(0x7CFC00AA,string);
	}
	if(pickupid == divenorth)
	{
		GameTextForPlayer(playerid,"~r~~h~Congrats!~n~~w~You finished the challenge.",6000,0);
		PlayerPlaySound(playerid,1149,0,0,0);
		GivePlayerWeapon(playerid,24,5000);
		GivePlayerWeapon(playerid,26,5000);
		GivePlayerWeapon(playerid,28,5000);
		GivePlayerWeapon(playerid,35,10);
		GivePlayerWeapon(playerid,18,10);
		SendClientMessage(playerid,0x7CFC00AA,"Reward: Weapons, RPG and Molotovs.");
    	new string[64];
    	new name[32];
    	GetPlayerName(playerid,name,128);
    	format(string, sizeof(string), "%s finished the challenge: Dive!",name);
		SendClientMessageToAll(0x7CFC00AA,string);
	}
	if(pickupid == divemiddle)
	{
		GameTextForPlayer(playerid,"~r~~h~Congrats!~n~~w~You finished the challenge.",6000,0);
		PlayerPlaySound(playerid,1149,0,0,0);
		GivePlayerWeapon(playerid,24,5000);
		GivePlayerWeapon(playerid,26,5000);
		GivePlayerWeapon(playerid,28,5000);
		GivePlayerWeapon(playerid,35,10);
		GivePlayerWeapon(playerid,18,10);
		SendClientMessage(playerid,0x7CFC00AA,"Reward: Weapons, RPG and Molotovs.");
   		new string[64];
   		new name[32];
   		GetPlayerName(playerid,name,128);
   		format(string, sizeof(string), "%s finished the challenge: Dive!",name);
		SendClientMessageToAll(0x7CFC00AA,string);
	}
	if(pickupid == parachute)
	{
		GivePlayerWeapon(playerid,46,1);
		DestroyPickup(parachute);
		SetTimer("paraspawn",5000,false);
	}
	if(pickupid == LV)
	{
		PlayerPlaySound(playerid, 1057, 0.0, 0.0, 0.0);
		SetPlayerInterior(playerid,0);
		SetPlayerPos(playerid,2026.4869,1342.3506,10.8203);
		SetSpawnInfo(playerid,-1,GetPlayerSkin(playerid),2026.4869,1342.3506,10.8203,270,0,0,0,0,0,0);
 		SendClientMessage(playerid,0x7CFC00AA,"* Las Venturas.");
		SetPlayerFacingAngle(playerid,270);
	}
	return 1;
}
public OnPlayerCommandText(playerid, cmdtext[])
{
	if (strcmp("/dive", cmdtext, true, 10) == 0)
	{
		SendClientMessage(playerid,0x7CFC00AA,"(Tip)Zoom out your camera and perform slow maneuvers!");
		GameTextForPlayer(playerid,"~r~~h~Get down",5000,6);
		ResetPlayerWeapons(playerid);
		GivePlayerWeapon(playerid,46,1);
		SetPlayerInterior (playerid, 0);
		SetPlayerHealth(playerid,100);
		SetPlayerArmour(playerid,100);
		SetPlayerPos(playerid,1867.1786,-146.1077,2465.8916);
		SetPlayerFacingAngle(playerid,270);
		SetCameraBehindPlayer(playerid);
		SetSpawnInfo(playerid,-1,GetPlayerSkin(playerid),1867.1786,-146.1077,2465.8916,270,46,1,0,0,0,0);
		return 1;
	}
	return 0;
}
