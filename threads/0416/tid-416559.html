    }

    if(pLanguage[playerid] == 1)
    {
        lan = 1;
    }

    if(pLanguage[playerid] == 2)
    {
        lan = 2;
    }

    if(pLanguage[playerid] == 3)
    {
        lan = 3;
    }
    return lan;
}
