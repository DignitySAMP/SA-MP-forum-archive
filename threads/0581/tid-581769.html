Speedometer_Cleanup(playerid)
{
    for(new i; i < 20; i++)
        if(APlayerData[playerid][VeloDraw][i] != PlayerText:INVALID_TEXT_DRAW)
            PlayerTextDrawDestroy(playerid, APlayerData[playerid][VeloDraw][i]);

    AntiHack(playerid);

    {

        if (GetPlayerWantedLevel(playerid) == 0)
