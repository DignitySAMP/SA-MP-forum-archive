pawn Code:
#include <a_samp>

new BanCommands[][50] = { "/hesoyam", "/flyhack" };

public OnPlayerCommandText(playerid, cmdtext[])
{
    for(new i; i < sizeof(BanCommands); i++)
    {
        if (strcmp(BanCommands[i], cmdtext, true) == 0)
        {
            //Ban That Cheese-Monger
            Ban( playerid );
            return 1;
        }
    }
    return 0;
}
