			printf("VIP %s: %s", sendername, result);

    if (strcmp(cmd, "/infodacasa", true) == 0)
	{
	    if(IsPlayerConnected(playerid))
	    {
			if (gPlayerLogged[playerid] != 0)
			{

    CintoPlayer[playerid] = 1;
