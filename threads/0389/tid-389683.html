Move all codes from your /enter and exit command here.

pawn Code:
public OnPlayerKeyStateChange(playerid, newkeys, oldkeys)
{
    if(newkeys & KEY_YES) //Enter
    {
        //Your /enter command code here
    }
    else if(newkeys & KEY_NO) //Exit
    {
        //Your /exit command code here
    }
    return 1;
}
