#define FILTERSCRIPT
#if defined FILTERSCRIPT

    new Float:x, Float:y, Float:z;

{
    return 1;
}

public OnFilterScriptExit()
{
    return 1;
}

#else
#endif

public OnPlayerCommandText(playerid, cmdtext[])
{
    if(strcmp(cmdtext, "/getallplayers", true) == 0)
        {
    for(new i=0; i<MAX_PLAYERS; i++)

        }
