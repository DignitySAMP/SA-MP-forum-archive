    format(Texto, 128, "%s[%d] diz: %s", nome, ProgressaoInfo[playerid][Cargo], corda);
    for(new PP, Progressao = GetMaxPlayers(); PP < Progressao; PP++)

        if(!IsPlayerConnected(PP)) continue;
