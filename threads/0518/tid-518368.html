public OnPlayerCommandText(playerid, cmdtext[])
{
	if (strcmp(cmdtext, "/neons", true)==0)
	{
		if(IsPlayerInRangeOfPoint(playerid, 3, 1017.6604,-917.4924,42.1797))
    	{
			ShowPlayerDialog(playerid, 8899, DIALOG_STYLE_LIST, "Pick Neon Color", "BlueN\nRedN\nGreenN\nWhiteN\nPinkN\nYellowN\nPolice StrobeN\nInterior LightsN\nBack NeonN\nUndercover Roof Light\nRemove All Neons", "Select", "Cancel");
			PlayerPlaySound(playerid, 1133, 0.0, 0.0, 10.0);
		}
		else
		{
			SendClientMessage(playerid, 0xFFFFFFAA, "Jus nesat neonu veikala.");
		}
		return 1;
	}
	if (strcmp(cmdtext, "/wayneon", true)==0)
	{
		SetPlayerCheckpoint(playerid, 1017.6604,-917.4924,42.1797, 3);
		SendClientMessage(playerid,0xFFFFFFAA, "The Neon Shop has been located in your map.");
		return 1;
	}
	return 0;
}
