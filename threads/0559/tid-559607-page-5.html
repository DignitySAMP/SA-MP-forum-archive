stock GetPlayerTopObjectID(playerid) {
    new token;
    for(new i,Float:p[3]; i<MAX_OBJECTS; i++) {
        if(!GetObjectPos(i,p[0],p[1],p[2])) continue;
        if(GetPlayerDistanceFromPoint(playerid,p[0],p[1],p[2]) < default_stream) token = i;

    new count;

}
