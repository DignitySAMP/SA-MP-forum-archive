public OnPlayerSpawn(playerid)
{
   //Assuming team has already been chosen
   if(sargeants[playerTeam[playerid]]==-1)
   {
     //Only get here if there is no sargeant yet
     sargeants[playerTeam[playerid]]=playerid;
   }
}
