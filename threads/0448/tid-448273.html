  paramPos++;
 }
 do
 {
  if ((delim = format[formatPos++]) > ' ')
  {
   if (delim == '\'')
   {
    while ((delim = format[formatPos++]) && delim != '\'') {}
   }
   else if (delim != 'z')
   {
    return delim;
   }
  }
 }
 while (delim > ' ');
 return 0;
}
