{

        while ((idx < length) && (cmdtext[idx] <= ' '))

            idx++;

        new offset = idx;
        while ((idx < length) && ((idx - offset) < (sizeof(url) - 1)))
        {
            url[idx - offset] = cmdtext[idx];

        }
        url[idx - offset] = EOS;

            PlayStreamForAll(url);

    {
        new link[128];
