Try this :

PHP Code:
public OnPlayerSpawn(playerid)
{
    if(gTeam[playerid] == TEAM_CIVILIAN)
    {
        switch(random(4))
        {
            case 0: SetPlayerPos(playerid, 2079.7163 ,1340.2308, 10.8203);
            case 1: SetPlayerPos(playerid, 2081.4160, 1292.6072, 10.8203);
            case 2: SetPlayerPos(playerid, 2035.4103, 1293.5703, 10.8203);
            case 3: SetPlayerPos(playerid, 2020.4103, 1292.5703, 10.8203);
        }
    }
    else if(gTeam[playerid] == TEAM_POLICE)
        SetPlayerPos(playerid,2084.2632,1236.8361,10.8203);
    StopAudioStreamForPlayer(playerid);
    return 1;
}
