public OnPlayerEditObject(playerid, playerobject, objectid, response, Float:fX, Float:fY, Float:fZ, Float:fRotX, Float:fRotY, Float:fRotZ)
{
	new string[128];
	if(!IsValidObject(objectid)) return 0;
	if(GetPVarInt(playerid, "voEditing") == 1)
	{
	    new vehid = voEditVeh[playerid];
	 	MoveObject(objectid, fX, fY, fZ, 10.0, fRotX, fRotY, fRotZ);
	 	if(response == EDIT_RESPONSE_FINAL)
		{
			new Float:ofx, Float:ofy, Float:ofz, Float:ofaz;
	  		new Float:finalx, Float:finaly;
	        new Float:px, Float:py, Float:pz, Float:roz;
	        GetVehiclePos(PlayerVehicleInfo[playerid][vehid][pvId], px, py, pz);
	        GetVehicleZAngle(PlayerVehicleInfo[playerid][vehid][pvId], roz);
	        ofx = fX-px;
	        ofy = fY-py;
	        ofz = fZ-pz;
	        ofaz = fRotZ-roz;
	        finalx = ofx*floatcos(roz, degrees)+ofy*floatsin(roz, degrees);
	        finaly = -ofx*floatsin(roz, degrees)+ofy*floatcos(roz, degrees);
	        AttachObjectToVehicle(PlayerVehicleInfo[playerid][vehid][pvtID][voEditSlot[playerid]], PlayerVehicleInfo[playerid][vehid][pvId], finalx, finaly, ofz, fRotX, fRotY, ofaz);
	        PlayerVehicleInfo[playerid][vehid][pvtPosX][voEditSlot[playerid]] = finalx;
	        PlayerVehicleInfo[playerid][vehid][pvtPosY][voEditSlot[playerid]] = finaly;
	        PlayerVehicleInfo[playerid][vehid][pvtPosZ][voEditSlot[playerid]] = ofz;
	        PlayerVehicleInfo[playerid][vehid][pvtRotX][voEditSlot[playerid]] = fRotX;
	        PlayerVehicleInfo[playerid][vehid][pvtRotY][voEditSlot[playerid]] = fRotY;
	        PlayerVehicleInfo[playerid][vehid][pvtRotZ][voEditSlot[playerid]] = ofaz;
			format(string, sizeof(string), " Anda telah mengedit posisi vehicle object slot %d pada kendaraan anda .", voEditSlot[playerid]);
			SendClientMessage(playerid, COLOR_WHITE, string);
			DeletePVar(playerid, "voEditing");
			voEditSlot[playerid] = 0;
			voEditVeh[playerid] = 0;
			return 1;
		}
	  	else if(response == EDIT_RESPONSE_CANCEL)
		{
			AttachObjectToVehicle(PlayerVehicleInfo[playerid][vehid][pvtID][voEditSlot[playerid]],
				PlayerVehicleInfo[playerid][vehid][pvId], PlayerVehicleInfo[playerid][vehid][pvtPosX][voEditSlot[playerid]],
				PlayerVehicleInfo[playerid][vehid][pvtPosY][voEditSlot[playerid]], PlayerVehicleInfo[playerid][vehid][pvtPosZ][voEditSlot[playerid]],
				PlayerVehicleInfo[playerid][vehid][pvtRotX][voEditSlot[playerid]], PlayerVehicleInfo[playerid][vehid][pvtRotY][voEditSlot[playerid]],
				PlayerVehicleInfo[playerid][vehid][pvtRotZ][voEditSlot[playerid]]
			);
			format(string, sizeof(string), " Anda batal mengedit posisi vehicle object slot %d pada kendaraan anda.", voEditSlot[playerid]);
			SendClientMessage(playerid, COLOR_WHITE, string);
			DeletePVar(playerid, "voEditing");
			voEditSlot[playerid] = 0;
			voEditVeh[playerid] = 0;
			return 1;
		}
	}
        return 0;

stock LoadPlayerVehicles(playerid)
{
	new playername[MAX_PLAYER_NAME];
	GetPlayerName(playerid, playername, sizeof(playername));
	printf(" Loading %s's vehicles.", playername);
	for(new v = 0; v < MAX_PLAYERVEHICLES; v++)
    {
        if(PlayerVehicleInfo[playerid][v][pvModelId] != 0 && PlayerVehicleInfo[playerid][v][pvImpounded] == 0)
        {
            PlayerCars++;
			new carcreated = CreateVehicle(PlayerVehicleInfo[playerid][v][pvModelId], PlayerVehicleInfo[playerid][v][pvPosX], PlayerVehicleInfo[playerid][v][pvPosY], PlayerVehicleInfo[playerid][v][pvPosZ], PlayerVehicleInfo[playerid][v][pvPosAngle],PlayerVehicleInfo[playerid][v][pvColor1], PlayerVehicleInfo[playerid][v][pvColor2], -1);
			PlayerVehicleInfo[playerid][v][pvId] = carcreated;
			VehicleFuel[carcreated] = PlayerVehicleInfo[playerid][v][pvFuel];
			SetVehicleNumberPlate(carcreated, "BKG");
			SetVehicleHealth(carcreated, PlayerVehicleInfo[playerid][v][pvHealth]);
			for(new vo = 0; vo < MAX_BODY; vo++)
   			{
				if(PlayerVehicleInfo[playerid][v][pvtModelID][vo] != 0 && PlayerVehicleInfo[playerid][v][pvtAttached][vo] != 0)
				{
                	PlayerVehicleInfo[playerid][v][pvtID][vo] = CreateObject(PlayerVehicleInfo[playerid][v][pvtModelID][vo],0,0,0,0,0,0);
		   			AttachObjectToVehicle(PlayerVehicleInfo[playerid][v][pvtID][vo], PlayerVehicleInfo[playerid][v][pvId], PlayerVehicleInfo[playerid][v][pvtPosX][vo], PlayerVehicleInfo[playerid][v][pvtPosY][vo], PlayerVehicleInfo[playerid][v][pvtPosZ][vo], PlayerVehicleInfo[playerid][v][pvtRotX][vo], PlayerVehicleInfo[playerid][v][pvtRotY][vo], PlayerVehicleInfo[playerid][v][pvtRotZ][vo]);
				}
			}
			
			if(PlayerVehicleInfo[playerid][v][pvLocked] == 1) LockPlayerVehicle(playerid, PlayerVehicleInfo[playerid][v][pvId]);
        }
        else if(PlayerVehicleInfo[playerid][v][pvModelId] == 0 && PlayerVehicleInfo[playerid][v][pvImpounded] != 0)
            PlayerVehicleInfo[playerid][v][pvImpounded] = 0;
    }
    LoadAllPlayerVehicleMods(playerid);
}

stock UnloadPlayerVehicles(playerid)
{
    new playername[MAX_PLAYER_NAME];
	GetPlayerName(playerid, playername, sizeof(playername));
	printf(" Unloading %s's vehicles.", playername);
	for(new v = 0; v < MAX_PLAYERVEHICLES; v++)
    {
        if(PlayerVehicleInfo[playerid][v][pvId] != INVALID_PLAYER_VEHICLE_ID && PlayerVehicleInfo[playerid][v][pvImpounded] == 0)
        {
            //new Float:x, Float:y, Float:z, Float:angle;

            //GetVehicleZAngle(PlayerVehicleInfo[playerid][v][pvId], angle);
            //UpdatePlayerVehicleParkPosition(playerid, v, x, y, z, angle);
            PlayerCars--;
            if(LockStatus[PlayerVehicleInfo[playerid][v][pvId]] != 0) LockStatus[PlayerVehicleInfo[playerid][v][pvId]] = 0;

 			VehicleFuel[PlayerVehicleInfo[playerid][v][pvId]] = 100.0;
			PlayerVehicleInfo[playerid][v][pvHealth]=1000.0;

				if(PlayerVehicleInfo[playerid][v][pvtModelID][vo] != 0)

	    		PlayerVehicleInfo[playerid][v][pvAllowedPlayerId] = INVALID_PLAYER_ID;
			}
        }
    }
}
