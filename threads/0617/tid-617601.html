forward ProxDetector(Float:radi, playerid, string[],col1,col2,col3,col4,col5);
public ProxDetector(Float:radi, playerid, string[],col1,col2,col3,col4,col5)
{
	if(IsPlayerConnected(playerid))
	{
		new Float:posx, Float:posy, Float:posz;
		new Float:oldposx, Float:oldposy, Float:oldposz;
		new Float:tempposx, Float:tempposy, Float:tempposz;
		GetPlayerPos(playerid, oldposx, oldposy, oldposz);
		for(new i = 0; i < MAX_PLAYERS; i++)
		{
			if(IsPlayerConnected(i) && (GetPlayerVirtualWorld(playerid) == GetPlayerVirtualWorld(i)))
			{
				GetPlayerPos(i, posx, posy, posz);
				tempposx = (oldposx -posx);
				tempposy = (oldposy -posy);
				tempposz = (oldposz -posz);
				if (((tempposx < radi/16) && (tempposx > -radi/16)) && ((tempposy < radi/16) && (tempposy > -radi/16)) && ((tempposz < radi/16) && (tempposz > -radi/16)))
				{
					SendClientMessage(i, col1, string);
				}
				else if (((tempposx < radi/8) && (tempposx > -radi/8)) && ((tempposy < radi/8) && (tempposy > -radi/8)) && ((tempposz < radi/8) && (tempposz > -radi/8)))
				{
					SendClientMessage(i, col2, string);
				}
				else if (((tempposx < radi/4) && (tempposx > -radi/4)) && ((tempposy < radi/4) && (tempposy > -radi/4)) && ((tempposz < radi/4) && (tempposz > -radi/4)))
				{
					SendClientMessage(i, col3, string);
				}
				else if (((tempposx < radi/2) && (tempposx > -radi/2)) && ((tempposy < radi/2) && (tempposy > -radi/2)) && ((tempposz < radi/2) && (tempposz > -radi/2)))
				{
					SendClientMessage(i, col4, string);
				}
				else if (((tempposx < radi) && (tempposx > -radi)) && ((tempposy < radi) && (tempposy > -radi)) && ((tempposz < radi) && (tempposz > -radi)))
				{
					SendClientMessage(i, col5, string);
				}
			}
			else
			{
				SendClientMessage(i, col1, string);
			}
		}
	}
	return 1;
}

CMD:tarifa(playerid, params[])
{
    if(PlayerInfo[playerid][Trabajotaxi] == 1)
    {
        new precio, id, string[128];
        new name[MAX_PLAYER_NAME], string2[24+MAX_PLAYER_NAME], nameid[MAX_PLAYER_NAME], string3[24+MAX_PLAYER_NAME];
        GetPlayerName(playerid, name, sizeof(name));
        GetPlayerName(playerid, nameid, sizeof(nameid));
        
        if (sscanf(params, "ui", id, precio))
            return SendClientMessage(playerid, 0xFFFFFFFF, "Usa: /tarifa <id> <precio>");

        if (!IsPlayerConnected(id))
            return SendClientMessage(playerid, 0xFFFFFFFF, "Jugador no estб {24610A}conectado!");

        if (id == playerid)
            return SendClientMessage(playerid, -1, "No puedes ofrecerte una tarifa a tн mismo");

        format(string, sizeof(string2), "%s te ofrecio una tarifa de {24610A}%d {FFFFFF}escribe {24610A}/aceptar tarifa para {FFFFFF}aceptar", precio,name);
        SendClientMessage(id, -1, string2);
        format(string, sizeof(string3), "Le ofreciste una tarifa a %s de {24610A}%d ", precio,nameid);
        SendClientMessage(playerid, -1, string3);
        Ofreciotarifa[id] = playerid;
        Preciotarifa[id] = precio;
    }
    return 1;
}

CMD:aceptar(playerid, params[])
{
    new tarifa[32];

    if(!sscanf(params, "s[32]",tarifa))
    {
        if(strcmp(tarifa, "tarifa",true) == 0)
        {
            if (Ofreciotarifa[playerid] == -1)
                return SendClientMessage(playerid, -1, "Nadie te ofreciу una tarifa");
            if(IsPlayerConnected(Ofreciotarifa[playerid]))
            {
                SendClientMessage(playerid, -1, "Has aceptado la tarifa");
                GivePlayerMoney(Ofreciotarifa[playerid], Preciotarifa[playerid]);
                GivePlayerMoney(playerid, -Preciotarifa[playerid]);
                Ofreciotarifa[playerid] = -1;
            } else SendClientMessage(playerid, -1, "El jugador no esta conectado");
        }
    } else SendClientMessage(playerid, -1, "/aceptar [tarifa]");
    return 1;
}
