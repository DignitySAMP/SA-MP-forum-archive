pawn Code:
stock vehicleLock(vehicleid, lock=1)
{
    playercar[vehicleid][car_locked] = lock;
    for(new playerid; playerid < MAX_PLAYERS; playerid++)
    {
      SetVehicleParamsForPlayer(vehicleid, playerid, 0, lock);
    }
    return 1;
}
