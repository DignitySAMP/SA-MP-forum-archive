stock ApplyAnim(playerid,animlib[],animname[], Float:Speed, looping, lockx, locky, lockz, lp)
{
    IsPlayerUsingAnims[playerid] = 1;
    ApplyAnimation(playerid,
					animlib,
					animname,
					Speed,
					looping,
					lockx,
					locky,
					lockz,
					lp);
	GameTextForPlayer(playerid, "~w~PRESS ~r~ Any key ~w~TO STOP ANIM", 3000, 3);
   	return true;
}

}
