stock GetLocation(Float:x, Float:y, zone[], len)
{
    for(new i = 0; i != sizeof(aLocations); i++ )
    {
        if(x >= aLocations[i][Coordinates][0] && x <= aLocations[i][Coordinates][3] && y >= aLocations[i][Coordinates][1] && \
        y <= aLocations[i][Coordinates][4]) return format(zone, len, aLocations[i][Name], 0);
