{
	new
	  vehicle,
		Float:x,

		Float:z,
		Float:angle;

	if ( GetPlayerState( playerid ) == PLAYER_STATE_DRIVER )

	#endif

	#endif
	#if !defined IGNORE_VEHICLE_DELETION

	#endif
	return 1;
}
