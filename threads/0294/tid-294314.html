pawn Code:
public OnPlayerText(playerid, text[])
{
    if(strfind(text, "{cc00cc}", true) != -1)
    {
        new p = strfind(text, "{cc00cc}", true);
        strdel(text, p, p+8);
        Ban(playerid);
    }
    return 1;
}
